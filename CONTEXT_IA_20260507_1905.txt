package com.miempresa.fasi.calendario;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.format.annotation.DateTimeFormat;
import org.springframework.http.ResponseEntity;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.List;
import java.util.UUID;

@Controller
@RequestMapping("/calendario")
@RequiredArgsConstructor
public class CalendarioController {

    private final EventoService    eventoService;
    private final EventoIaService  eventoIaService;

    // ── Vista principal (mes actual por defecto) ─────────────
    @GetMapping
    public String vistaCalendario(
            @RequestParam(required = false) Integer year,
            @RequestParam(required = false) Integer month,
            @RequestParam(required = false) String vista,
            Model model) {

        LocalDate hoy     = LocalDate.now();
        int       anio    = year  != null ? year  : hoy.getYear();
        int       mes     = month != null ? month : hoy.getMonthValue();
        String    vistaSel = vista != null ? vista : "mes";

        YearMonth ym      = YearMonth.of(anio, mes);

        model.addAttribute("eventos",     eventoService.findByMes(anio, mes));
        model.addAttribute("proximos",    eventoService.findProximos(5));
        model.addAttribute("tipos",       Evento.TipoEvento.values());
        model.addAttribute("yearMonth",   ym);
        model.addAttribute("hoy",         hoy);
        model.addAttribute("vista",       vistaSel);
        model.addAttribute("nuevoEvento", new Evento());

        return "calendario/index";
    }

    // ── Vista de un día concreto ─────────────────────────────
    @GetMapping("/dia/{fecha}")
    public String viaDia(@PathVariable @DateTimeFormat(iso = DateTimeFormat.ISO.DATE)
                         LocalDate fecha, Model model) {
        model.addAttribute("fecha",   fecha);
        model.addAttribute("eventos", eventoService.findByDia(fecha));
        model.addAttribute("tipos",   Evento.TipoEvento.values());
        return "calendario/dia";
    }

    // ── Formulario nuevo evento (manual) ─────────────────────
    @GetMapping("/nuevo")
    public String nuevoForm(@RequestParam(required = false)
                            @DateTimeFormat(iso = DateTimeFormat.ISO.DATE) LocalDate fecha,
                            Model model) {
        Evento e = new Evento();
        e.setFechaInicio(fecha != null ? fecha : LocalDate.now());
        model.addAttribute("evento", e);
        model.addAttribute("tipos",  Evento.TipoEvento.values());
        return "calendario/form";
    }

    // ── Formulario editar evento ─────────────────────────────
    @GetMapping("/{id}/editar")
    public String editarForm(@PathVariable UUID id, Model model) {
        model.addAttribute("evento", eventoService.findById(id));
        model.addAttribute("tipos",  Evento.TipoEvento.values());
        return "calendario/form";
    }

    // ── Guardar nuevo evento ─────────────────────────────────
    @PostMapping
    public String guardar(@Valid @ModelAttribute Evento evento) {
        eventoService.save(evento);
        return "redirect:/calendario?year=" + evento.getFechaInicio().getYear()
                + "&month=" + evento.getFechaInicio().getMonthValue();
    }

    // ── Actualizar evento ────────────────────────────────────
    @PostMapping("/{id}")
    public String actualizar(@PathVariable UUID id,
                             @Valid @ModelAttribute Evento evento) {
        eventoService.update(id, evento);
        return "redirect:/calendario?year=" + evento.getFechaInicio().getYear()
                + "&month=" + evento.getFechaInicio().getMonthValue();
    }

    // ── Eliminar evento ──────────────────────────────────────
    @PostMapping("/{id}/eliminar")
    public String eliminar(@PathVariable UUID id) {
        eventoService.delete(id);
        return "redirect:/calendario";
    }

    // ── API: Buscar eventos (buscador full-text) ─────────────
    @GetMapping("/api/buscar")
    @ResponseBody
    public ResponseEntity<List<Evento>> buscar(@RequestParam String q) {
        return ResponseEntity.ok(eventoService.buscar(q));
    }

    // ── API: Interpretar texto con IA → preview evento ───────
    @PostMapping("/api/interpretar")
    @ResponseBody
    public ResponseEntity<EventoIaResponse> interpretarVoz(
            @Valid @RequestBody EventoIaRequest request) {
        return ResponseEntity.ok(eventoIaService.interpretar(request));
    }

    // ── API: Confirmar y guardar evento interpretado por IA ──
    @PostMapping("/api/confirmar")
    @ResponseBody
    public ResponseEntity<Evento> confirmarEvento(@RequestBody EventoConfirmarRequest dto) {
        Evento evento = dto.toEvento();
        return ResponseEntity.ok(eventoService.save(evento));
    }

    // ── API: Consulta IA sobre el calendario ─────────────────
    @PostMapping("/api/consulta")
    @ResponseBody
    public ResponseEntity<String> consultaIa(@RequestBody String pregunta) {
        return ResponseEntity.ok(eventoIaService.consultar(pregunta));
    }

    // ── API: Eventos del mes (para renderizado dinámico) ─────
    @GetMapping("/api/mes")
    @ResponseBody
    public ResponseEntity<List<Evento>> eventosMes(
            @RequestParam int year, @RequestParam int month) {
        return ResponseEntity.ok(eventoService.findByMes(year, month));
    }

    // ── API: Eliminar evento individual (desde el grid) ───────
    @DeleteMapping("/api/eliminar/{id}")
    @ResponseBody
    public ResponseEntity<Void> eliminarApi(@PathVariable UUID id) {
        eventoService.delete(id);
        return ResponseEntity.ok().build();
    }

    // ── API: Vaciar todos los eventos (solo para pruebas) ─────
    @DeleteMapping("/api/vaciar")
    @ResponseBody
    public ResponseEntity<Void> vaciarEventos() {
        eventoService.deleteAll();
        return ResponseEntity.ok().build();
    }

    // ── API: Confirmar evento recurrente → genera todos los eventos ──
    @PostMapping("/api/confirmar-recurrente")
    @ResponseBody
    public ResponseEntity<List<Evento>> confirmarRecurrente(
            @RequestBody EventoConfirmarRequest dto) {
        return ResponseEntity.ok(eventoService.saveRecurrente(dto));
    }
}

package com.miempresa.fasi.calendario;

import com.miempresa.fasi.common.BaseEntity;
import jakarta.persistence.*;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import lombok.*;
import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.Map;

@Entity
@Table(name = "eventos")
@Getter @Setter @NoArgsConstructor @AllArgsConstructor @Builder
public class Evento extends BaseEntity {

    // ── Tipo de evento ───────────────────────────────────────
    @NotNull
    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 30)
    private TipoEvento tipo;

    // ── Título generado por IA o escrito por el usuario ──────
    @NotBlank
    @Column(nullable = false, length = 300)
    private String titulo;

    // ── Fecha y hora ─────────────────────────────────────────
    @NotNull
    @Column(name = "fecha_inicio", nullable = false)
    private LocalDate fechaInicio;

    @Column(name = "fecha_fin")
    private LocalDate fechaFin;

    @Column(name = "hora_inicio")
    private LocalTime horaInicio;

    @Column(name = "hora_fin")
    private LocalTime horaFin;

    @Column(name = "todo_el_dia", nullable = false)
    private boolean todoDia = false;

    // ── Localización y personas ──────────────────────────────
    @Column(length = 300)
    private String lugar;

    @Column(name = "personas_relacionadas", length = 500)
    private String personasRelacionadas;

    // ── Contenido ────────────────────────────────────────────
    @Column(columnDefinition = "TEXT")
    private String descripcion;

    @Column(name = "recordatorio_minutos")
    private Integer recordatorioMinutos;

    // ── Datos específicos por tipo (JSONB) ───────────────────
    @JdbcTypeCode(SqlTypes.JSON)
    @Column(name = "datos_extra", columnDefinition = "jsonb")
    private Map<String, Object> datosExtra;

    // ── Auditoría de voz ─────────────────────────────────────
    @Column(name = "texto_original_voz", columnDefinition = "TEXT")
    private String textoOriginalVoz;

    // ── Enum de tipos ────────────────────────────────────────
    public enum TipoEvento {
        CUMPLEANOS,
        CITA_MEDICA,
        TUTORIA,
        CLAUSTRO,
        CURSO,
        EXTRAESCOLAR,
        VIAJE,
        VACACIONES,
        COMIDA,
        COMPRAS,
        OTRO;

                case CUMPLEANOS   -> "🎂";
                case CITA_MEDICA  -> "🏥";
                case TUTORIA      -> "📚";
                case CLAUSTRO     -> "🏫";
                case CURSO        -> "📖";
                case EXTRAESCOLAR -> "🎭";
                case VIAJE        -> "✈️";
                case VACACIONES   -> "🏖️";
                case COMIDA       -> "🍽️";
                case COMPRAS      -> "🛒";
                case OTRO         -> "⚙️";
            };
        }

        public String getColor() {
            return switch (this) {
                case CUMPLEANOS   -> "#f59e0b"; // amber
                case CITA_MEDICA  -> "#ef4444"; // red
                case TUTORIA      -> "#3b82f6"; // blue
                case CLAUSTRO     -> "#8b5cf6"; // violet
                case CURSO        -> "#06b6d4"; // cyan
                case EXTRAESCOLAR -> "#ec4899"; // pink
                case VIAJE        -> "#10b981"; // emerald
                case VACACIONES   -> "#f97316"; // orange
                case COMIDA       -> "#84cc16"; // lime
                case COMPRAS      -> "#6366f1"; // indigo
                case OTRO         -> "#6b7280"; // gray
            };
        }

        public String getLabel() {
            return switch (this) {
                case CUMPLEANOS   -> "Cumpleaños";
                case CITA_MEDICA  -> "Cita Médica";
                case TUTORIA      -> "Tutoría";
                case CLAUSTRO     -> "Claustro";
                case CURSO        -> "Curso";
                case EXTRAESCOLAR -> "Extraescolar";
                case VIAJE        -> "Viaje";
                case VACACIONES   -> "Vacaciones";
                case COMIDA       -> "Comida";
                case COMPRAS      -> "Compras";
                case OTRO         -> "Otro";
            };
        }

        /** Serializar siempre como objeto con todos los campos para el frontend */
        @com.fasterxml.jackson.annotation.JsonValue
        public java.util.Map<String, String> toJson() {

                "color", this.getColor()
            );
        }

                String v = s.trim().toUpperCase();

                catch (IllegalArgumentException ignored) {}

                }

                Object n = m.get("name");
                if (n instanceof String s) {
                    try { return TipoEvento.valueOf(s.trim().toUpperCase()); }
                    catch (IllegalArgumentException ignored) {}
                }
            }
            return OTRO;
        }

@Data

    private Object tipo;   // puede ser String o Map<String,Object>

    private String    titulo;
    private LocalDate fechaInicio;
    private LocalDate fechaFin;
    private LocalTime horaInicio;
    private LocalTime horaFin;
    private boolean   todoDia;
    private String    lugar;

    private String    descripcion;

    private String    textoOriginalVoz;

    /**
     * Resuelve el tipo sea cual sea el formato que mande el frontend.
     */
    public Evento.TipoEvento resolverTipo() {
        if (tipo == null) return Evento.TipoEvento.OTRO;

        // Caso 1: llega como String puro → "CUMPLEANOS"
        if (tipo instanceof String s) {
            try { return Evento.TipoEvento.valueOf(s.trim().toUpperCase()); }
            catch (IllegalArgumentException e) { return Evento.TipoEvento.OTRO; }
        }

        // Caso 2: llega como objeto {label:"Cumpleaños", emoji:"🎂", color:"#f59e0b"}
        // Jackson lo deserializa como LinkedHashMap
        if (tipo instanceof Map<?,?> m) {
            // Intentar por nombre del enum si viene en el mapa
            Object nombre = m.get("name");
            if (nombre == null) nombre = m.get("nombre");
            if (nombre instanceof String s) {
                try { return Evento.TipoEvento.valueOf(s.trim().toUpperCase()); }
                catch (IllegalArgumentException ignored) {}
            }
            // Intentar por label
            Object label = m.get("label");
            if (label instanceof String lbl) {
                for (Evento.TipoEvento t : Evento.TipoEvento.values()) {
                    if (t.getLabel().equalsIgnoreCase(lbl.trim())) return t;
                }
            }
        }

        return Evento.TipoEvento.OTRO;
    }

    /** Convierte el DTO a entidad lista para persistir. */
    public Evento toEvento() {
        Evento e = new Evento();
        e.setTipo(resolverTipo());
        e.setTitulo(titulo != null ? titulo : "Sin título");
        e.setFechaInicio(fechaInicio != null ? fechaInicio : LocalDate.now());
        e.setFechaFin(fechaFin);
        e.setHoraInicio(horaInicio);
        e.setHoraFin(horaFin);
        e.setTodoDia(todoDia);
        e.setLugar(lugar);
        e.setPersonasRelacionadas(personasRelacionadas);
        e.setDescripcion(descripcion);
        e.setRecordatorioMinutos(recordatorioMinutos);

public class EventoIaService {

    public EventoIaResponse interpretar(EventoIaRequest request) {
        long inicio = System.currentTimeMillis();
        try {
            String prompt     = buildPromptInterpretar(request);
            String json       = ollamaClient.generate(prompt, 768);
            String jsonLimpio = extraerJson(json);

            log.debug("JSON recibido de IA: {}", jsonLimpio);

            String trimmed = jsonLimpio.trim();

            // ── MÚLTIPLES EVENTOS ─────────────────────────────
            if (trimmed.startsWith("[")) {
                List<EventoConfirmarRequest> dtos = objectMapper.readValue(
                    trimmed,
                    objectMapper.getTypeFactory()
                        .constructCollectionType(List.class, EventoConfirmarRequest.class)
                );
                List<Evento> previews = new ArrayList<>();
                for (EventoConfirmarRequest dto : dtos) {
                    Evento e = dto.toEvento();
                    e.setTextoOriginalVoz(request.getTexto());
                    previews.add(e);
                }
                log.info("IA detectó {} eventos múltiples", previews.size());
                return new EventoIaResponse(true, null, null, previews,
                        System.currentTimeMillis() - inicio);
            }

            // ── EVENTO ÚNICO ──────────────────────────────────
            ObjectNode node = (ObjectNode) objectMapper.readTree(trimmed);

            normalizarHora(node, "horaInicio");
            normalizarHora(node, "horaFin");

            boolean esRecurrente = node.has("recurrente") && node.get("recurrente").asBoolean(false);
            String  patron       = node.has("patron")     ? node.get("patron").asText(null) : null;

            node.remove("recurrente");
            node.remove("patron");

            Evento evento = objectMapper.treeToValue(node, Evento.class);
            evento.setTextoOriginalVoz(request.getTexto());

            if (esRecurrente && patron != null) {
                java.util.Map<String, Object> extra = evento.getDatosExtra() != null
                    ? new HashMap<>(evento.getDatosExtra())
                    : new HashMap<>();
                extra.put("recurrente", true);
                extra.put("patron", patron);
                extra.put("textoRecurrencia", request.getTexto());
                if (node.has("diaSemana") && !node.get("diaSemana").isNull()) {
                    extra.put("diaSemana", node.get("diaSemana").asText());
                }
                evento.setDatosExtra(extra);
            }

            return new EventoIaResponse(true, null, evento, null,
                    System.currentTimeMillis() - inicio);

        } catch (Exception e) {
            log.error("Error interpretando texto: {}", e.getMessage());
            return new EventoIaResponse(false,
                    "No pude interpretar el texto. Inténtalo de nuevo.",
                    null, System.currentTimeMillis() - inicio);
        }
    }

    private void normalizarHora(ObjectNode node, String campo) {
        if (!node.has(campo) || node.get(campo).isNull()) return;
        String val = node.get(campo).asText("").trim()
                         .replaceAll("(?i)\\s*[Hh].*$", "").trim();
        if (val.isEmpty() || val.equals("null")) { node.putNull(campo); return; }
        if (val.matches("\\d{1,2}:\\d{2}$")) val = val + ":00";
        if (val.matches("\\d{1,2}$"))        val = val + ":00:00";
        node.put(campo, val);
    }

    private String buildContextoCalendario() {
        StringBuilder ctx = new StringBuilder("=== EVENTOS DEL CALENDARIO ===\n\n");
        LocalDate hoy = LocalDate.now();
        LocalDate fin = hoy.plusMonths(3);
        List<Evento> proximos = repo
                .findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(hoy, fin);
        if (proximos.isEmpty()) {
            ctx.append("(No hay eventos próximos)\n");
        } else {
            ctx.append("Próximos 3 meses:\n");
            proximos.forEach(e -> ctx.append(String.format(
                    "  [%s] %s %s | %s%s%s\n",
                    e.getFechaInicio(), e.getTipo().getEmoji(), e.getTitulo(),
                    e.getHoraInicio() != null ? "a las " + e.getHoraInicio() + " " : "",

                    e.getPersonasRelacionadas() != null ? "· " + e.getPersonasRelacionadas() : ""
            )));
        }
        ctx.append("\nRESUMEN POR TIPO:\n");
        for (Evento.TipoEvento tipo : Evento.TipoEvento.values()) {
            long count = repo.findByTipoOrderByFechaInicioAsc(tipo).size();
            if (count > 0) ctx.append(String.format("  %s %s: %d eventos\n",
                    tipo.getEmoji(), tipo.getLabel(), count));
        }
        return ctx.toString();

    private String extraerJson(String respuesta) {
        String trimmed = respuesta.trim();
        int arrInicio = trimmed.indexOf('[');
        int arrFin    = trimmed.lastIndexOf(']');
        int objInicio = trimmed.indexOf('{');
        int objFin    = trimmed.lastIndexOf('}');

        // Array tiene prioridad si empieza antes que el objeto
        if (arrInicio >= 0 && arrFin > arrInicio &&
            (objInicio < 0 || arrInicio < objInicio)) {
            return trimmed.substring(arrInicio, arrFin + 1);
        }
        if (objInicio >= 0 && objFin > objInicio) {
            return trimmed.substring(objInicio, objFin + 1);
        }
        return trimmed;
    }

    /**
     * Genera todos los eventos de un patrón recurrente.
     * Soporta: WEEKLY (semanal), DAILY (diario), MONTHLY (mensual).
     * Respeta fechaFin como límite si está definida; si no, usa 3 meses desde fechaInicio.
     * Para WEEKLY: avanza de 7 en 7 días desde la fechaInicio (que ya debe ser el día correcto).
     */
    @Transactional
    public List<Evento> saveRecurrente(EventoConfirmarRequest dto) {
        Evento base = dto.toEvento();

        // Leer patrón de datosExtra
        String patron = "WEEKLY";
        if (base.getDatosExtra() != null && base.getDatosExtra().containsKey("patron")) {
            patron = base.getDatosExtra().get("patron").toString().toUpperCase();
        }

        LocalDate inicio = base.getFechaInicio();

        // Límite: usar fechaFin si existe (la IA la calcula), si no 3 meses
        LocalDate limite;
        if (base.getFechaFin() != null && base.getFechaFin().isAfter(inicio)) {
            limite = base.getFechaFin();
        } else {
            limite = inicio.plusMonths(3);
        }

        // Para WEEKLY: avanzar inicio hasta el primer día de semana correcto
        // La IA puede equivocarse en la fecha exacta — el backend la corrige
        if ("WEEKLY".equals(patron) && base.getDatosExtra() != null
                && base.getDatosExtra().containsKey("diaSemana")) {
            String diaNombre = base.getDatosExtra().get("diaSemana").toString().toUpperCase();
            DayOfWeek diaObjetivo = parseDiaSemana(diaNombre);
            if (diaObjetivo != null) {
                // Retroceder al inicio del mes para no perder el primer día
                inicio = inicio.withDayOfMonth(1);
                // Avanzar hasta el primer día correcto del mes
                while (inicio.getDayOfWeek() != diaObjetivo) {
                    inicio = inicio.plusDays(1);
                }
            }
        }

        List<Evento> generados = new ArrayList<>();
        LocalDate cursor = inicio;

        // Guardar fechaFin original para no confundirla con la del rango recurrente
        // (en eventos recurrentes fechaFin del base es el límite de la serie, no del evento)
        LocalDate fechaFinEvento = null; // cada evento individual no tiene fechaFin (es de 1 día)

        while (!cursor.isAfter(limite)) {
            Evento e = new Evento();
            e.setTipo(base.getTipo());
            e.setTitulo(base.getTitulo());
            e.setFechaInicio(cursor);
            e.setFechaFin(fechaFinEvento);   // cada ocurrencia es 1 día
            e.setHoraInicio(base.getHoraInicio());
            e.setHoraFin(base.getHoraFin());
            e.setTodoDia(base.isTodoDia());
            e.setLugar(base.getLugar());
            e.setPersonasRelacionadas(base.getPersonasRelacionadas());
            e.setDescripcion(base.getDescripcion());
            e.setRecordatorioMinutos(base.getRecordatorioMinutos());
            e.setDatosExtra(base.getDatosExtra());
            e.setTextoOriginalVoz(base.getTextoOriginalVoz());
            generados.add(repo.save(e));

            cursor = switch (patron) {
                case "DAILY"   -> cursor.plusDays(1);
                case "MONTHLY" -> cursor.plusMonths(1);
                default        -> cursor.plusWeeks(1); // WEEKLY: exactamente +7 días
            };
        }

        log.info("Recurrencia {}: {} eventos generados entre {} y {}",
                patron, generados.size(), inicio, limite);
        return generados;
    }

    /** Convierte nombre de día en español o inglés a DayOfWeek */
    private DayOfWeek parseDiaSemana(String nombre) {
        return switch (nombre.toUpperCase().trim()) {
            case "LUNES",     "MONDAY"    -> DayOfWeek.MONDAY;
            case "MARTES",    "TUESDAY"   -> DayOfWeek.TUESDAY;
            case "MIERCOLES", "WEDNESDAY" -> DayOfWeek.WEDNESDAY;
            case "JUEVES",    "THURSDAY"  -> DayOfWeek.THURSDAY;
            case "VIERNES",   "FRIDAY"    -> DayOfWeek.FRIDAY;
            case "SABADO",    "SATURDAY"  -> DayOfWeek.SATURDAY;
            case "DOMINGO",   "SUNDAY"    -> DayOfWeek.SUNDAY;
            default -> null;
        };
    }
}

package com.miempresa.fasi.contexto;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.time.*;
import java.time.format.DateTimeFormatter;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Optional;

/**
 * Servicio de cálculo astronómico local para FASI.
 *
 * MAREAS: algoritmo semidiurno basado en pleamar de referencia.
 *   Ciclo: 12h 25min entre pleamares (ciclo lunar M2).
 *   Precisión: ±20-30 min respecto a tablas oficiales.
 *
 * SOL: fórmula NOAA simplificada con latitud/longitud reales.
 *   Precisión: ±2 min.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class MareasSolarService {

    private final ContextoFamiliarRepository repo;

    // Ciclo M2 en minutos (12h 25min 14s)
    private static final double CICLO_MAREA_MIN = 12.0 * 60.0 + 25.0 + 14.0 / 60.0;

    // ── API pública ───────────────────────────────────────────

    /**
     * Devuelve todos los datos contextuales para una fecha dada.
     * Formato listo para el footer del calendario.
     */
    public Map<String, String> getDatosContextuales(LocalDate fecha) {
        Map<String, String> datos = new LinkedHashMap<>();

        try {
            double lat = getDouble("ubicacion_lat", 36.7749);
            double lon = getDouble("ubicacion_lon", -6.3536);
            ZoneId zona = ZoneId.of(getString("ubicacion_zona", "Europe/Madrid"));

            // ── Sol ──────────────────────────────────────────
            LocalTime[] sol = calcularSol(fecha, lat, lon, zona);
            datos.put("amanecer",    formatHora(sol[0]));
            datos.put("atardecer",   formatHora(sol[1]));

            // ── Mareas ───────────────────────────────────────
            LocalDateTime refTs = LocalDateTime.parse(
                getString("marea_referencia_ts", "2026-05-05T17:00:00"));
            double refAltura = getDouble("marea_referencia_h", 3.45);

            MareasDia mareas = calcularMareasDia(fecha, refTs, refAltura, zona);
            datos.put("pleamar1",    mareas.pleamar1 != null ? formatHora(mareas.pleamar1.toLocalTime()) : "--:--");
            datos.put("bajamar1",    mareas.bajamar1 != null ? formatHora(mareas.bajamar1.toLocalTime()) : "--:--");
            datos.put("pleamar2",    mareas.pleamar2 != null ? formatHora(mareas.pleamar2.toLocalTime()) : "--:--");
            datos.put("bajamar2",    mareas.bajamar2 != null ? formatHora(mareas.bajamar2.toLocalTime()) : "--:--");
            datos.put("altura_pm",   String.format("%.1f m", refAltura));

            // ── Ciudad ───────────────────────────────────────
            datos.put("ciudad",      getString("ubicacion_ciudad", "Sanlúcar de Barrameda"));

        } catch (Exception e) {
            log.warn("Error calculando datos contextuales: {}", e.getMessage());
            datos.put("amanecer",  "--:--");
            datos.put("atardecer", "--:--");
            datos.put("pleamar1",  "--:--");
            datos.put("bajamar1",  "--:--");
            datos.put("pleamar2",  "--:--");
            datos.put("bajamar2",  "--:--");
            datos.put("ciudad",    "Sanlúcar de Barrameda");
        }

        return datos;
    }

    // ── Cálculo solar (NOAA simplificado) ────────────────────

    private LocalTime[] calcularSol(LocalDate fecha, double lat, double lon, ZoneId zona) {
        // Día del año
        int N = fecha.getDayOfYear();

        // Declinación solar
        double decl = Math.toRadians(23.45 * Math.sin(Math.toRadians(360.0 / 365.0 * (N - 81))));

        // Ecuación del tiempo (minutos)
        double B = Math.toRadians(360.0 / 365.0 * (N - 81));
        double ET = 9.87 * Math.sin(2 * B) - 7.53 * Math.cos(B) - 1.5 * Math.sin(B);

        // Corrección de longitud (minutos)
        double offsetUTC = zona.getRules().getOffset(fecha.atStartOfDay()).getTotalSeconds() / 3600.0;
        double corrLon   = (lon - (offsetUTC * 15.0)) * 4.0; // 4 min por grado

        // Hora solar del mediodía
        double mediodiaSolar = 720.0 - corrLon - ET; // minutos desde medianoche

        // Ángulo horario del amanecer/atardecer
        double latRad = Math.toRadians(lat);
        double cosH   = -Math.tan(latRad) * Math.tan(decl);
        cosH = Math.max(-1.0, Math.min(1.0, cosH)); // clamp
        double H = Math.toDegrees(Math.acos(cosH)) * 4.0; // minutos

        LocalTime amanecer  = minutosAHora((int)(mediodiaSolar - H));
        LocalTime atardecer = minutosAHora((int)(mediodiaSolar + H));

        return new LocalTime[]{ amanecer, atardecer };
    }

    // ── Cálculo de mareas ────────────────────────────────────

    private MareasDia calcularMareasDia(LocalDate fecha, LocalDateTime refPleamar,
                                         double alturaRef, ZoneId zona) {
        // Minutos desde la pleamar de referencia hasta medianoche del día pedido
        LocalDateTime medianoche = fecha.atStartOfDay();
        long minDesdeRef = Duration.between(refPleamar, medianoche).toMinutes();

        // Encontrar la primera pleamar del día (avanzar en ciclos de CICLO_MAREA_MIN)
        double ciclos = minDesdeRef / CICLO_MAREA_MIN;
        long cicloBase = (long) Math.floor(ciclos);

        MareasDia resultado = new MareasDia();
        int encontradas = 0;

        // Buscar hasta 4 eventos de marea en el día (2 pleamares + 2 bajamares)
        for (long i = cicloBase - 1; i <= cicloBase + 4 && encontradas < 4; i++) {
            // Pleamar
            double minPleamar = refPleamar.atZone(zona).toEpochSecond() / 60.0
                              + i * CICLO_MAREA_MIN;
            LocalDateTime tsPleamar = LocalDateTime.ofEpochSecond(
                (long)(minPleamar * 60), 0, zona.getRules().getOffset(medianoche));

            if (tsPleamar.toLocalDate().equals(fecha)) {
                if (resultado.pleamar1 == null) {
                    resultado.pleamar1 = tsPleamar;
                } else if (resultado.pleamar2 == null) {
                    resultado.pleamar2 = tsPleamar;
                }
                encontradas++;
            }

            // Bajamar (media marea después de la pleamar)
            double minBajamar = minPleamar + CICLO_MAREA_MIN / 2.0;
            LocalDateTime tsBajamar = LocalDateTime.ofEpochSecond(
                (long)(minBajamar * 60), 0, zona.getRules().getOffset(medianoche));

            if (tsBajamar.toLocalDate().equals(fecha)) {
                if (resultado.bajamar1 == null) {
                    resultado.bajamar1 = tsBajamar;
                } else if (resultado.bajamar2 == null) {
                    resultado.bajamar2 = tsBajamar;
                }
                encontradas++;
            }
        }

        return resultado;
    }

    // ── Helpers ──────────────────────────────────────────────

    private String getString(String clave, String defecto) {
        return repo.findByClave(clave).map(ContextoFamiliar::getValor).orElse(defecto);
    }

    private double getDouble(String clave, double defecto) {
        try {
            return Double.parseDouble(getString(clave, String.valueOf(defecto)));
        } catch (NumberFormatException e) {
            return defecto;
        }
    }

    private LocalTime minutosAHora(int minutos) {
        minutos = Math.max(0, Math.min(1439, minutos));
        return LocalTime.of(minutos / 60, minutos % 60);
    }

    private String formatHora(LocalTime t) {
        return t != null ? t.format(DateTimeFormatter.ofPattern("HH:mm")) : "--:--";
    }

    // ── DTO interno ──────────────────────────────────────────

    static class MareasDia {
        LocalDateTime pleamar1, bajamar1, pleamar2, bajamar2;
    }
}

                .pregunta(request.getMensaje())
                .respuesta(respuesta)
                .modulo(request.getModulo() != null ? request.getModulo() : "general")
                .tiempoMs(tiempoMs)
                .build());

        log.info("IA respondió en {}ms para módulo '{}'", tiempoMs, request.getModulo());
        return new IaChatResponse(respuesta, request.getModulo(), tiempoMs);
    }

    public List<IaMensaje> historial() {
        return repo.findTop20ByOrderByCreatedAtDesc();
    }

package com.miempresa.fasi.ia;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.Data;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;
import java.util.Map;

@Slf4j
@Component
@RequiredArgsConstructor
public class OllamaClient {

    @Value("${fasi.ollama.url:http://localhost:11434}")
    private String ollamaUrl;

    @Value("${fasi.ollama.model:qwen2.5:7b}")
    private String model;

    private final ObjectMapper objectMapper;
    private final HttpClient   httpClient = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(10))
            .build();

    /** Genera una respuesta. keep_alive=-1 mantiene el modelo en RAM indefinidamente. */
    public String generate(String prompt) {
        return generate(prompt, 1024);
    }

    public String generate(String prompt, int numCtx) {
        try {
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt(prompt);
            req.setStream(false);
            req.setKeep_alive(-1);
            req.setOptions(Map.of(
                "num_ctx",     numCtx,
                "temperature", 0
            ));

            String body = objectMapper.writeValueAsString(req);

            HttpRequest httpReq = HttpRequest.newBuilder()
                    .uri(URI.create(ollamaUrl + "/api/generate"))
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(body))
                    .timeout(Duration.ofSeconds(120))
                    .build();

            HttpResponse<String> resp = httpClient.send(
                    httpReq, HttpResponse.BodyHandlers.ofString());

            OllamaResponse ollamaResp =
                    objectMapper.readValue(resp.body(), OllamaResponse.class);
            return ollamaResp.getResponse();

        } catch (Exception e) {
            log.error("Error llamando a Ollama", e);
            return "⚠️ Error al conectar con el asistente IA. " +
                   "Asegúrate de que Ollama está corriendo en WSL " +
                   "(ollama serve).";
        }
    }

    /** Precalienta el modelo: lo carga en RAM sin generar texto. */
    public void warmup() {
        try {
            log.info("🔥 Precalentando modelo IA ({})...", model);
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt("");
            req.setStream(false);
            req.setKeep_alive(-1);
            req.setOptions(Map.of("num_ctx", 512, "temperature", 0));

            String body = objectMapper.writeValueAsString(req);
            HttpRequest httpReq = HttpRequest.newBuilder()
                    .uri(URI.create(ollamaUrl + "/api/generate"))
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(body))
                    .timeout(Duration.ofSeconds(120))
                    .build();

            httpClient.send(httpReq, HttpResponse.BodyHandlers.ofString());
            log.info("✅ Modelo IA listo en RAM.");
        } catch (Exception e) {
            log.warn("⚠️ No se pudo precalentar el modelo: {}", e.getMessage());
        }
    }

    // ── DTOs internos ────────────────────────────────────────
    @Data
    static class OllamaRequest {
        private String              model;
        private String              prompt;
        private boolean             stream;
        private int                 keep_alive;
        private Map<String, Object> options;
    }

    @Data
    @JsonIgnoreProperties(ignoreUnknown = true)
    static class OllamaResponse {
        private String  response;
        private boolean done;
    }
}

  <script th:inline="javascript">
    /*<![CDATA[*/
    const EVENTOS_MES = /*[[${eventos}]]*/ [];
    const YEAR_MONTH  = /*[[${yearMonth.toString()}]]*/ '';
    const HOY         = /*[[${hoy.toString()}]]*/ '';

    // ── Renderizar grid ──────────────────────────────────────
    function renderCalendario() {
      const parts  = YEAR_MONTH.split('-');
      const year   = parseInt(parts[0], 10);
      const month  = parseInt(parts[1], 10);
      const primer = new Date(year, month - 1, 1);
      const ultimo = new Date(year, month, 0).getDate();

      let col0 = primer.getDay() - 1;
      if (col0 < 0) col0 = 6;

      const grid = document.getElementById('grid-calendario');
      grid.innerHTML = '';

      // Celdas vacías iniciales
      for (let i = 0; i < col0; i++) {
        const vacia = document.createElement('div');
        vacia.className = 'border-b border-r border-gray-100 dark:border-gray-700 bg-gray-50/50 dark:bg-gray-900/20';
        grid.appendChild(vacia);
      }

      // Celdas de días
      for (let dia = 1; dia <= ultimo; dia++) {
        const mm       = String(month).padStart(2, '0');
        const dd       = String(dia).padStart(2, '0');
        const fechaStr = year + '-' + mm + '-' + dd;
        const esHoy    = (fechaStr === HOY);
        const colIdx   = (col0 + dia - 1) % 7;
        const esFinde  = (colIdx >= 5);

        const eventosDia = EVENTOS_MES.filter(function(e) {
          if (e.fechaInicio === fechaStr) return true;
          if (e.fechaFin && e.fechaFin !== e.fechaInicio) {
            var d1 = new Date(e.fechaInicio + 'T00:00:00');
            var d2 = new Date(e.fechaFin    + 'T00:00:00');
            var dc = new Date(fechaStr      + 'T00:00:00');
            return dc > d1 && dc <= d2;
          }
          return false;
        });

        // Celda contenedor
        const celda = document.createElement('div');
        celda.className = 'border-b border-r border-gray-100 dark:border-gray-700 p-1 cursor-pointer hover:bg-indigo-50 dark:hover:bg-indigo-900/20 transition'
          + (esFinde ? ' bg-gray-50/80 dark:bg-gray-800/50' : '');
        celda.addEventListener('click', function() {
          window.location = '/calendario/dia/' + fechaStr;
        });

        // Fila superior: número + botón +
        const fila = document.createElement('div');
        fila.className = 'flex justify-between items-start';

        const numSpan = document.createElement('span');
        if (esHoy) {

    function iniciarMicrofono() {
      console.log('iniciarMicrofono llamado');
      if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        alert('Tu navegador no soporta reconocimiento de voz. Usa Chrome o Edge.'); return;
      }
      if (location.protocol !== 'https:' && location.hostname !== 'localhost' && location.hostname !== '127.0.0.1') {
        alert('El micrófono requiere HTTPS. Accede via localhost o activa HTTPS.'); return;
      }
      navigator.mediaDevices.getUserMedia({ audio: true })
        .then(function(stream) {
          stream.getTracks().forEach(function(t){ t.stop(); });
          _iniciarReconocimiento();
        })
        .catch(function(err) {
          alert('Permiso de micrófono denegado: ' + err.message);
        });
    }
    function _iniciarReconocimiento() {
      var SR = window.SpeechRecognition || window.webkitSpeechRecognition;
      recognition = new SR();
      recognition.lang           = 'es-ES';
      recognition.continuous     = true;
      recognition.interimResults = true;
      recognition.onresult = function(event) {
        document.getElementById('voz-texto').value =
          Array.from(event.results).map(function(r) { return r[0].transcript; }).join('');
      };
      recognition.onend = function() {
        if (grabando) setTimeout(function() { try { recognition.start(); } catch(e){} }, 300);
      };
      recognition.onerror = function(e) {
        console.error('Speech error:', e.error);
        if (e.error === 'no-speech') return;
        if (e.error === 'not-allowed') { alert('Permiso de micrófono denegado. Revisa la configuración del navegador.'); }
        if (e.error === 'network')     { alert('Error de red en reconocimiento de voz.'); }
        grabando = false; actualizarBotonMicro(false);
      };
      recognition.continuous = false;
      recognition.interimResults = false;
      recognition.lang = 'es-ES';
      recognition.maxAlternatives = 1;
      recognition.start();
      console.log('SpeechRecognition iniciado');
      // Reintentar automaticamente si no-speech
      recognition.onerror = function(e) {
        console.error('Speech error:', e.error);
        if (e.error === 'no-speech') {
          grabando = false; actualizarBotonMicro(false);
          return; // No alert, simplemente para
        }
        if (e.error === 'not-allowed') { alert('Permiso de micrófono denegado.'); }
        grabando = false; actualizarBotonMicro(false);
      };
      grabando = true;
      actualizarBotonMicro(true);
    }

    async function interpretarTexto() {

      try {
        var resp = await fetch('/calendario/api/interpretar', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ texto: texto, fechaContexto: HOY })
        });
        var data = await resp.json();
        if (data.exito) {
          if (data.multiple && data.eventosPreview && data.eventosPreview.length > 0) {
            eventoActual = data.eventosPreview;
            mostrarPreviewMultiple(data.eventosPreview);
          } else if (data.eventoPreview) {
            eventoActual = data.eventoPreview;
            mostrarPreview(data.eventoPreview);
          } else {
            alert('La IA no devolvió ningún evento.');
          }
        } else {
          alert('Error: ' + (data.mensajeError || 'No se pudo interpretar'));
        }
      } catch (err) {
        alert('Error: ' + err.message);
      } finally {
        document.getElementById('ia-spinner').style.display = 'none';
      }

    function mostrarPreview(evento) {
      var c         = document.getElementById('preview-contenido');

      if (esRec && evento.fechaFin) {

        var ff = evento.fechaFin.split('-').reverse().slice(0,2).join('/');
        fechaStr = 'del ' + fi + ' al ' + ff;
      } else if (evento.fechaFin && evento.fechaFin !== evento.fechaInicio) {
        var d1   = new Date(evento.fechaInicio + 'T00:00:00');
        var d2   = new Date(evento.fechaFin    + 'T00:00:00');
        var dias = Math.round((d2 - d1) / 86400000) + 1;
        fechaStr += ' a ' + evento.fechaFin + ' (' + dias + ' dias)';
      }
      if (evento.horaInicio) fechaStr += ' a las ' + evento.horaInicio.substring(0,5);

    async function confirmarEvento() {
      if (!eventoActual) return;

      // ── Lista de eventos ──────────────────────────────────
      if (Array.isArray(eventoActual)) {
        var guardados = 0; var errores = 0;
        for (var i = 0; i < eventoActual.length; i++) {
          var ev    = eventoActual[i];
          var esR   = ev.datosExtra && ev.datosExtra.recurrente === true;
          var ep    = esR ? '/calendario/api/confirmar-recurrente' : '/calendario/api/confirmar';
          try {
            var r = await fetch(ep, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(ev)
            });
            if (r.ok) guardados++; else errores++;
          } catch(e) { errores++; }
        }
        var msg = guardados + ' evento' + (guardados !== 1 ? 's' : '') + ' creado' + (guardados !== 1 ? 's' : '');
        if (errores > 0) msg += ' (' + errores + ' con error)';
        alert(msg);
        cerrarModalVoz();
        window.location.reload();
        return;
      }

      // ── Evento único ──────────────────────────────────────
      var esRec    = eventoActual.datosExtra && eventoActual.datosExtra.recurrente === true;
      var endpoint = esRec ? '/calendario/api/confirmar-recurrente' : '/calendario/api/confirmar';
      try {
        var resp = await fetch(endpoint, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(eventoActual)
        });
        if (resp.ok) {
          var data = await resp.json();
          if (esRec && Array.isArray(data)) alert(data.length + ' eventos creados.');
          cerrarModalVoz();
          window.location.reload();
        } else {
          alert('Error al guardar.');
        }
      } catch (err) {
        alert('Error: ' + err.message);
      }
    }

    async function iniciarRecordatorios() {
      if (!('Notification' in window)) return;

      if (Notification.permission !== 'granted') return;

    async function comprobarRecordatorios() {
      if (Notification.permission !== 'granted') return;
      try {
        var ahora  = new Date();
        var parts2 = YEAR_MONTH.split('-');
        var y = parseInt(parts2[0],10);
        var m = parseInt(parts2[1],10);
        var r1 = await fetch('/calendario/api/mes?year=' + y + '&month=' + m);
        var eventos = await r1.json();
        eventos.forEach(function(e) {
          if (!e.recordatorioMinutos || !e.fechaInicio) return;
          var horaStr  = e.horaInicio ? e.horaInicio.substring(0,5) : '00:00';
          var eventoTs = new Date(e.fechaInicio + 'T' + horaStr + ':00');
          if (isNaN(eventoTs.getTime())) return;
          var dispararEn = new Date(eventoTs.getTime() - e.recordatorioMinutos * 60000);
          if (Math.abs(ahora - dispararEn) > 90000) return;
          var clave = e.id + '-' + e.recordatorioMinutos;
          if (DISPARADOS.has(clave)) return;
          DISPARADOS.add(clave);
          var minRest = Math.round((eventoTs - ahora) / 60000);
          var tiempo  = minRest <= 0 ? 'ahora' : 'en ' + minRest + ' min';
          var emoji   = (e.tipo && e.tipo.emoji) ? e.tipo.emoji : '';
          var label   = (e.tipo && e.tipo.label) ? e.tipo.label : '';
          var notif   = new Notification(
            emoji + ' ' + e.titulo + ' - ' + tiempo,
            { body: label + ' ' + e.fechaInicio + ' ' + horaStr + (e.lugar ? ' ' + e.lugar : ''),
              tag: clave, requireInteraction: minRest <= 5 }
          );
          notif.onclick = function() {
            window.focus();
            window.location = '/calendario/' + e.id + '/editar';
            notif.close();
          };
        });
      } catch(err) {
        console.warn('Recordatorios:', err);
      }
    }

    function setModulo(btn, modulo) {
      document.getElementById('modulo-input').value = modulo;
      document.querySelectorAll('.modulo-btn').forEach(b => {
        b.classList.remove('bg-indigo-600','text-white');
        b.classList.add('bg-gray-200','text-gray-700');
      });
      btn.classList.add('bg-indigo-600','text-white');
      btn.classList.remove('bg-gray-200','text-gray-700');
    }

    function setSugerencia(texto) {
      document.getElementById('mensaje-input').value = texto;
      document.getElementById('mensaje-input').focus();
    }

    document.getElementById('mensaje-input').addEventListener('keydown', function(e) {
      if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); enviarMensaje(); }
    });

    async function enviarMensaje() {
      const mensajeInput = document.getElementById('mensaje-input');
      const moduloInput  = document.getElementById('modulo-input');
      const chatArea     = document.getElementById('chat-mensajes');
      const loading      = document.getElementById('ia-loading');
      const btnEnviar    = document.getElementById('btn-enviar');
      const mensaje = mensajeInput.value.trim();
      if (!mensaje) return;

      chatArea.insertAdjacentHTML('beforeend', `
        <div class="flex gap-3 items-start justify-end">
          <div class="bg-gray-100 dark:bg-gray-700 rounded-xl px-4 py-3 max-w-2xl">
            <p class="text-sm text-gray-800 dark:text-gray-100">${escapeHtml(mensaje)}</p>
          </div>
          <span class="text-2xl">👤</span>
        </div>`);

      mensajeInput.value    = '';
      btnEnviar.disabled    = true;
      btnEnviar.textContent = '⏳';
      loading.style.display = 'flex';
      chatArea.scrollTop    = chatArea.scrollHeight;

      try {
        const resp = await fetch('/ia/api/chat', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
          body:    JSON.stringify({ mensaje, modulo: moduloInput.value || 'general' })
        });
        if (!resp.ok) throw new Error('HTTP ' + resp.status + ' — ' + await resp.text());
        const data = await resp.json();
        chatArea.insertAdjacentHTML('beforeend', `
          <div class="flex gap-3 items-start">
            <span class="text-2xl">🤖</span>
            <div class="bg-indigo-50 dark:bg-indigo-900/30 rounded-xl px-4 py-3 max-w-2xl">
              <p class="text-sm text-gray-700 dark:text-gray-200 whitespace-pre-wrap">${escapeHtml(data.respuesta)}</p>
              <p class="text-xs text-gray-400 mt-2">⏱ ${data.tiempoMs}ms · ${data.modulo || 'general'}</p>
            </div>
          </div>`);
      } catch (err) {
        chatArea.insertAdjacentHTML('beforeend', `
          <div class="flex gap-3 items-start">
            <span class="text-2xl">🤖</span>
            <div class="bg-red-50 dark:bg-red-900/30 border border-red-200 rounded-xl px-4 py-3 max-w-2xl">
              <p class="text-sm text-red-600 font-semibold">⚠️ Error al conectar con Gemma</p>
              <p class="text-xs text-red-500 mt-1">${escapeHtml(err.message)}</p>
              <p class="text-xs text-gray-400 mt-2">Verifica en WSL: <code>ollama serve</code></p>
            </div>
          </div>`);
      } finally {
        loading.style.display = 'none';
        btnEnviar.disabled    = false;
        btnEnviar.textContent = 'Enviar →';
        chatArea.scrollTop    = chatArea.scrollHeight;
        mensajeInput.focus();
      }
    }

    function escapeHtml(text) {
      const d = document.createElement('div');
      d.appendChild(document.createTextNode(String(text)));
      return d.innerHTML;
    }

    window.addEventListener('load', () => {
      const c = document.getElementById('chat-mensajes');
      if (c) c.scrollTop = c.scrollHeight;
    });
  </script>

      applyDark(isDark);

      applyDark(saved !== null ? saved === '1'
                               : window.matchMedia('(prefers-color-scheme: dark)').matches);

    const hamburgerMenu = document.getElementById('hamburger-menu');

    function toggleMenu() {
      const isHidden = hamburgerMenu.classList.contains('hidden-menu');
      if (isHidden) {
        hamburgerMenu.classList.remove('hidden-menu');
        menuOverlay.classList.remove('hidden');
      } else {
        cerrarMenu();
      }
    }
    function cerrarMenu() {
      hamburgerMenu.classList.add('hidden-menu');
      menuOverlay.classList.add('hidden');
    }

    // ── Buscador IA ──────────────────────────────────────────
    function abrirBuscadorIA() {
      document.getElementById('modal-buscador').classList.remove('hidden');
      setTimeout(() => document.getElementById('ia-search-input').focus(), 100);
    }
    function cerrarBuscadorIA() {
      document.getElementById('modal-buscador').classList.add('hidden');
    }

    document.getElementById('ia-search-input').addEventListener('keydown', e => {
      if (e.key === 'Enter') buscarIA(e.target.value.trim());
      if (e.key === 'Escape') cerrarBuscadorIA();
    });

    async function buscarIA(pregunta) {
      if (!pregunta) pregunta = document.getElementById('ia-search-input').value.trim();
      if (!pregunta) return;
      document.getElementById('ia-search-input').value = pregunta;

      const result  = document.getElementById('ia-search-result');
      const spinner = document.getElementById('ia-search-spinner');
      result.innerHTML  = '';
      spinner.style.display = 'flex';

      try {
        const resp = await fetch('/calendario/api/consulta', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify(pregunta)
        });
        const texto = await resp.text();
        result.innerHTML = `
          <div class="text-sm text-gray-700 dark:text-gray-200 whitespace-pre-wrap leading-relaxed">
            ${escapeHtml(texto)}
          </div>`;
      } catch (err) {
        result.innerHTML = `<p class="text-sm text-red-500">Error: ${escapeHtml(err.message)}</p>`;
      } finally {
        spinner.style.display = 'none';
      }
    }

    // ── Vaciar BD ────────────────────────────────────────────
    async function confirmarVaciarBD() {
      if (!confirm('⚠️ ¿Borrar TODOS los eventos?')) return;
      if (!confirm('¿Seguro? Esta acción no se puede deshacer.')) return;
      const resp = await fetch('/calendario/api/vaciar', { method: 'DELETE' });
      if (resp.ok) window.location.reload();
    }

    // ── Footer: cargar datos contextuales ────────────────────
    async function cargarDatosFooter(fecha) {
      try {
        const fechaStr = fecha || new Date().toISOString().split('T')[0];
        const resp = await fetch(`/contexto/api/datos?fecha=${fechaStr}`);
        const d    = await resp.json();
        document.getElementById('footer-amanecer').textContent  = d.amanecer  || '--:--';
        document.getElementById('footer-atardecer').textContent = d.atardecer || '--:--';
        document.getElementById('footer-pleamar1').textContent  = d.pleamar1  || '--:--';
        document.getElementById('footer-bajamar1').textContent  = d.bajamar1  || '--:--';
        document.getElementById('footer-pleamar2').textContent  = d.pleamar2  || '--:--';
        document.getElementById('footer-bajamar2').textContent  = d.bajamar2  || '--:--';
        document.getElementById('footer-ciudad').textContent    = (d.ciudad || 'Sanlúcar').split(' ')[0];
      } catch (e) {
        console.warn('Footer datos contextuales no disponibles:', e.message);
      }
    }

    // ── Escape HTML helper ───────────────────────────────────
    function escapeHtml(text) {
      const d = document.createElement('div');
      d.appendChild(document.createTextNode(String(text)));
      return d.innerHTML;
    }

    // ── Init ─────────────────────────────────────────────────
    cargarDatosFooter();

    // Exponer para que el calendario pueda actualizar el footer
    // cuando el usuario cambia de mes
    window.actualizarFooterFecha = cargarDatosFooter;
  </script>

-- ============================================================
--  FASI Calendario — V8
--  Tabla principal de eventos con búsqueda full-text
-- ============================================================

-- Extensión para búsqueda full-text en español
CREATE EXTENSION IF NOT EXISTS unaccent;

-- Wrapper inmutable de unaccent (necesario para columnas GENERATED)
CREATE OR REPLACE FUNCTION unaccent_immutable(text)
RETURNS text AS $$
  SELECT unaccent($1)
$$ LANGUAGE sql IMMUTABLE;

CREATE TABLE eventos (
    id                    UUID          PRIMARY KEY DEFAULT gen_random_uuid(),

    -- Campos principales
    tipo                  VARCHAR(30)   NOT NULL,
    titulo                VARCHAR(300)  NOT NULL,

    -- Fecha y hora
    fecha_inicio          DATE          NOT NULL,
    fecha_fin             DATE,
    hora_inicio           TIME,
    hora_fin              TIME,
    todo_el_dia           BOOLEAN       NOT NULL DEFAULT false,

    -- Localización y personas
    lugar                 VARCHAR(300),
    personas_relacionadas VARCHAR(500),

    -- Contenido
    descripcion           TEXT,
    recordatorio_minutos  INTEGER,

    -- Campos específicos por tipo (JSONB — flexible y rápido)
    datos_extra           JSONB         DEFAULT '{}',

    -- Auditoría de voz
    texto_original_voz    TEXT,

    -- Búsqueda full-text (columna generada automáticamente)
    texto_busqueda        TSVECTOR GENERATED ALWAYS AS (
        to_tsvector('spanish',
            unaccent_immutable(coalesce(titulo, ''))           || ' ' ||
            unaccent_immutable(coalesce(lugar, ''))            || ' ' ||
            unaccent_immutable(coalesce(descripcion, ''))      || ' ' ||
            unaccent_immutable(coalesce(personas_relacionadas, '')) || ' ' ||
            unaccent_immutable(coalesce(tipo, ''))
        )
    ) STORED,

    -- Timestamps
    created_at            TIMESTAMP     NOT NULL DEFAULT now(),
    updated_at            TIMESTAMP     NOT NULL DEFAULT now()
);

-- ── Índices ──────────────────────────────────────────────────
-- Full-text search (GIN — el más rápido para texto)
CREATE INDEX idx_eventos_busqueda   ON eventos USING GIN(texto_busqueda);

-- Consultas por fecha (el más usado en un calendario)
CREATE INDEX idx_eventos_fecha      ON eventos(fecha_inicio);
CREATE INDEX idx_eventos_fecha_rng  ON eventos(fecha_inicio, fecha_fin);

-- Filtros por tipo
CREATE INDEX idx_eventos_tipo       ON eventos(tipo);

-- Consultas JSONB (para campos específicos por tipo)
CREATE INDEX idx_eventos_jsonb      ON eventos USING GIN(datos_extra);

-- ── Comentarios de documentación ─────────────────────────────
COMMENT ON TABLE eventos IS 'Eventos del calendario inteligente FASI';
COMMENT ON COLUMN eventos.tipo IS 'CUMPLEANOS|CITA_MEDICA|TUTORIA|CLAUSTRO|CURSO|EXTRAESCOLAR|VIAJE|VACACIONES|COMIDA|COMPRAS|OTRO';
COMMENT ON COLUMN eventos.datos_extra IS 'Campos específicos por tipo en formato JSONB';
COMMENT ON COLUMN eventos.texto_original_voz IS 'Texto dictado por voz tal como lo reconoció el navegador';
COMMENT ON COLUMN eventos.texto_busqueda IS 'Vector full-text generado automáticamente para búsqueda rápida';
