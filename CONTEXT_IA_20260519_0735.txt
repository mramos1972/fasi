// ════════════════════════════════════════════════════════════
//  FASI Service Worker — Network-first para HTML, cache-first para estáticos
//  Versión: 20260514_170228
// ════════════════════════════════════════════════════════════

const VERSION    = '20260514_170228';
const CACHE_NAME = 'fasi-' + VERSION;
const STATIC_ASSETS = [
  '/manifest.json',
  '/images/icon-192.png',
  '/images/icon-512.png'
];

// ── Instalación: pre-cachear estáticos ──────────────────────
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(STATIC_ASSETS).catch(() => {}))
      .then(() => self.skipWaiting())   // activar inmediatamente
  );
});

// ── Activación: limpiar cachés viejas ───────────────────────
self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys()
      .then(keys => Promise.all(
        keys.filter(k => k !== CACHE_NAME).map(k => caches.delete(k))
      ))
      .then(() => self.clients.claim())  // tomar control de páginas abiertas
  );
});

// ── Fetch: estrategia híbrida según tipo de recurso ─────────
self.addEventListener('fetch', event => {
  const req = event.request;
  const url = new URL(req.url);

  // No interceptar peticiones que no son GET
  if (req.method !== 'GET') return;

  // No cachear APIs (siempre directo al servidor)
  if (url.pathname.startsWith('/calendario/api/')
   || url.pathname.startsWith('/ia/api/')
   || url.pathname.startsWith('/contexto/api/')
   || url.pathname.startsWith('/login')
   || url.pathname.startsWith('/logout')) {
    return;  // dejar que el navegador haga su petición normal
  }

  // ── HTML y navegación: network-first (siempre fresco) ────
  if (req.mode === 'navigate'
   || req.headers.get('accept')?.includes('text/html')) {
    event.respondWith(
      fetch(req)
        .then(resp => {
          // Guardar copia en caché por si luego hay offline
          const copy = resp.clone();
          caches.open(CACHE_NAME).then(c => c.put(req, copy));
          return resp;
        })
        .catch(() => caches.match(req))   // fallback offline
    );
    return;
  }

  // ── Estáticos (CSS, JS, imágenes): cache-first ───────────
  event.respondWith(
    caches.match(req).then(cached => {
      if (cached) return cached;
      return fetch(req).then(resp => {
        const copy = resp.clone();
        if (resp.status === 200) {
          caches.open(CACHE_NAME).then(c => c.put(req, copy));
        }
        return resp;
      });
    })
  );
});

// ── Permitir que la app fuerce un skipWaiting desde JS ──────
self.addEventListener('message', event => {
  if (event.data === 'SKIP_WAITING') self.skipWaiting();
});

    // ── Vista principal (mes actual por defecto) ─────────────
    @GetMapping
    public String vistaCalendario(
            @RequestParam(required = false) Integer year,
            @RequestParam(required = false) Integer month,
            @RequestParam(required = false) String vista,
            Model model) {

        LocalDate hoy     = LocalDate.now();
        int       anio    = year  != null ? year  : hoy.getYear();
        int       mes     = month != null ? month : hoy.getMonthValue();
        String    vistaSel = vista != null ? vista : "mes";

        YearMonth ym      = YearMonth.of(anio, mes);

        model.addAttribute("eventos",     eventoService.findByMes(anio, mes));
        model.addAttribute("proximos",    eventoService.findProximos(5));
        model.addAttribute("tipos",       Evento.TipoEvento.values());
        model.addAttribute("yearMonth",   ym);
        model.addAttribute("hoy",         hoy);
        model.addAttribute("vista",       vistaSel);
        model.addAttribute("nuevoEvento", new Evento());

        return "calendario/index";
    }

    private java.util.Set<String> parseEtiquetas(String raw) {
        if (raw == null || raw.isBlank()) return new java.util.HashSet<>();
        return java.util.Arrays.stream(raw.split(","))
                .map(String::trim)
                .map(String::toLowerCase)
                .filter(s -> !s.isEmpty())
                .map(s -> s.replaceAll("[áàä]", "a")
                           .replaceAll("[éèë]", "e")
                           .replaceAll("[íìï]", "i")
                           .replaceAll("[óòö]", "o")
                           .replaceAll("[úùü]", "u")
                           .replaceAll("[ñ]", "n"))
                .collect(java.util.stream.Collectors.toCollection(java.util.HashSet::new));
    }

    @PostMapping("/api/interpretar-preview")
    @ResponseBody
    public Map<String, Object> interpretarConPreview(@RequestBody Map<String, String> body) {
        String texto = body.get("texto");
        LocalDate hoy = LocalDate.now();

        log.info("📥 Interpretando con preview: {}", texto);

        // Intentar REGEX primero
        Optional<EventoParserRegex.EventoParseResult> resultadoRegex = regexParser.parse(texto, hoy);
        
        Evento evento;
        boolean esRecurrente;
        
        if (resultadoRegex.isPresent()) {
            log.info("⚡ REGEX exitoso");
            evento = resultadoRegex.get().evento();
            esRecurrente = resultadoRegex.get().esRecurrente();
        } else {
            log.info("⏩ REGEX no coincide → usando IA");
            EventoIaRequest request = new EventoIaRequest();
            request.setTexto(texto);
            request.setFechaContexto(hoy.toString());
            
            EventoIaResponse iaResponse = eventoIaService.interpretar(request);
            
            if (!iaResponse.isExito() || iaResponse.getEventoPreview() == null) {
                Map<String, Object> errorResponse = new HashMap<>();
                errorResponse.put("error", iaResponse.getMensajeError());
                return errorResponse;
            }
            
            evento = iaResponse.getEventoPreview();
            esRecurrente = evento.getDatosExtra() != null && 
                          Boolean.TRUE.equals(evento.getDatosExtra().get("recurrente"));
        }

        Map<String, Object> response = new HashMap<>();
        response.put("evento", evento);
        response.put("esRecurrente", esRecurrente);

        // Si es recurrente, generar vista previa de fechas
        if (esRecurrente) {
            List<LocalDate> fechas = eventoService.generarVistaPrevia(evento);
            response.put("totalEventos", fechas.size());
            response.put("primerasFechas", fechas.stream().limit(10).toList());

            // Generar descripción legible
            String descripcion = generarDescripcionPreview(fechas);
            response.put("descripcionPreview", descripcion);
        }

        return response;
    }
    
    private String generarDescripcionPreview(List<LocalDate> fechas) {
        if (fechas.isEmpty()) return "No se generarán eventos";
        
        StringBuilder sb = new StringBuilder();
        sb.append("Se crearán ").append(fechas.size()).append(" eventos:\n");
        
        int limite = Math.min(10, fechas.size());
        for (int i = 0; i < limite; i++) {
            LocalDate fecha = fechas.get(i);
            String diaSemana = switch (fecha.getDayOfWeek()) {
                case MONDAY -> "Lunes";
                case TUESDAY -> "Martes";
                case WEDNESDAY -> "Miércoles";
                case THURSDAY -> "Jueves";
                case FRIDAY -> "Viernes";
                case SATURDAY -> "Sábado";
                case SUNDAY -> "Domingo";
            };
            sb.append("• ").append(diaSemana).append(" ")
              .append(String.format("%02d/%02d/%d", fecha.getDayOfMonth(), fecha.getMonthValue(), fecha.getYear()))
              .append("\n");
        }
        
        if (fechas.size() > 10) {
            sb.append("... y ").append(fechas.size() - 10).append(" más");
        }
        
        return sb.toString();
    }

                catch (IllegalArgumentException ignored) {}
                for (TipoEvento t : values()) {
                    if (t.getLabel().equalsIgnoreCase(s.trim())) return t;

            if (value instanceof java.util.Map<?,?> m) {
                Object n = m.get("name");
                if (n instanceof String s) {
                    try { return TipoEvento.valueOf(s.trim().toUpperCase()); }
                    catch (IllegalArgumentException ignored) {}
                }
            }
            return OTRO;
        }

package com.miempresa.fasi.calendario;

import lombok.Data;
import java.time.LocalDate;
import java.time.LocalTime;
import java.util.HashMap;
import java.util.Map;

@Data
public class EventoConfirmarRequest {
    private Evento.TipoEvento tipo;
    private String            titulo;
    private LocalDate         fechaInicio;
    private LocalDate         fechaFin;
    private LocalTime         horaInicio;
    private LocalTime         horaFin;
    private boolean           todoDia;
    private String            lugar;
    private String            personasRelacionadas;
    private String            descripcion;
    private Integer           recordatorioMinutos;
    private Map<String, Object> datosExtra;
    private String            textoOriginalVoz;

    public Evento toEvento() {
        Evento e = new Evento();
        e.setTipo(tipo != null ? tipo : Evento.TipoEvento.OTRO);
        e.setTitulo(titulo);
        e.setFechaInicio(fechaInicio);
        e.setFechaFin(fechaFin);
        e.setHoraInicio(horaInicio);
        e.setHoraFin(horaFin);
        e.setTodoDia(todoDia);
        e.setLugar(lugar);
        e.setPersonasRelacionadas(personasRelacionadas);
        e.setDescripcion(descripcion);
        e.setRecordatorioMinutos(recordatorioMinutos);
        e.setTextoOriginalVoz(textoOriginalVoz);

        Map<String, Object> extra = datosExtra != null
            ? new HashMap<>(datosExtra)
            : new HashMap<>();

        // Si no tiene hora y no es todo el día → marcar pendiente
        if (horaInicio == null && !todoDia) {
            extra.put("pendienteHora", true);
        }

        e.setDatosExtra(extra.isEmpty() ? null : extra);
        return e;
    }
}

package com.miempresa.fasi.calendario;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.miempresa.fasi.ia.IaContextBuilder;
import com.miempresa.fasi.ia.OllamaClient;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.*;

@Slf4j
@Service
@RequiredArgsConstructor
public class EventoIaService {

    private final OllamaClient           ollamaClient;
    private final EventoRepository       repo;
    private final ObjectMapper           objectMapper;
    private final IaContextBuilder       contextBuilder;
    private final EventoParserRegex      parserRegex;

    private static final int CTX_INTERPRETAR   = 512;
    private static final int MAX_OUTPUT_TOKENS = 600;

    // ══════════════════════════════════════════════════════════
    //  SISTEMA HÍBRIDO: REGEX → IA
    // ══════════════════════════════════════════════════════════

    public EventoIaResponse interpretar(EventoIaRequest request) {
        long inicio = System.currentTimeMillis();
        LocalDate fechaContexto = request.getFechaContexto() != null && !request.getFechaContexto().isBlank()
            ? LocalDate.parse(request.getFechaContexto())
            : LocalDate.now();

        try {
            // ── PASO 1: Intentar REGEX (< 50ms) ──────────────────
            Optional<EventoParserRegex.EventoParseResult> resultado =
                parserRegex.parse(request.getTexto(), fechaContexto);

            if (resultado.isPresent()) {
                long elapsed = System.currentTimeMillis() - inicio;
                Evento evento = resultado.get().evento();
                boolean esRecurrente = resultado.get().esRecurrente();

                log.info("⚡ REGEX exitoso en {}ms (recurrente: {})", elapsed, esRecurrente);

                return new EventoIaResponse(true, null, evento, null, elapsed);
            }

            log.info("⏩ REGEX no coincide → usando IA");

            // ── PASO 2: Usar IA para casos complejos ─────────────
            return interpretarConIA(request, fechaContexto, inicio);

        } catch (Exception e) {
            log.error("❌ Error interpretando texto: {}", e.getMessage(), e);
            return new EventoIaResponse(false,
                    "No pude interpretar el texto. Inténtalo de nuevo.",
                    null, System.currentTimeMillis() - inicio);
        }
    }

    private EventoIaResponse interpretarConIA(EventoIaRequest request, LocalDate fechaContexto, long inicio) throws Exception {
        String prompt = construirPrompt(request.getTexto(), fechaContexto);
        String json   = ollamaClient.generateJson(prompt, CTX_INTERPRETAR, MAX_OUTPUT_TOKENS);

        log.info("📥 IA RAW: {}", json);

        String jsonLimpio = extraerJson(json);
        log.info("📋 IA LIMPIO: {}", jsonLimpio);

        JsonNode root = objectMapper.readTree(jsonLimpio);
        JsonNode eventosNode = root.has("eventos") ? root.get("eventos") : root;

        List<ObjectNode> nodos = new ArrayList<>();
        if (eventosNode.isArray()) {
            eventosNode.forEach(n -> { if (n.isObject()) nodos.add((ObjectNode) n); });
        } else if (eventosNode.isObject()) {
            nodos.add((ObjectNode) eventosNode);
        }

        if (nodos.isEmpty()) {
            throw new IllegalStateException("La IA no devolvió eventos en el JSON.");
        }

        List<Evento> eventos = new ArrayList<>();
        for (ObjectNode node : nodos) {
            normalizarHora(node, "horaInicio");
            normalizarHora(node, "horaFin");
            normalizarPersonas(node);

            boolean esRecurrente = node.has("recurrente") && node.get("recurrente").asBoolean(false);
            String  patron = node.has("patron") ? node.get("patron").asText(null) : null;

            node.remove("recurrente");
            node.remove("patron");

            Evento evento = objectMapper.treeToValue(node, Evento.class);
            evento.setTextoOriginalVoz(request.getTexto());

            if (esRecurrente && patron != null) {
                Map<String, Object> extra = evento.getDatosExtra() != null
                    ? new HashMap<>(evento.getDatosExtra())
                    : new HashMap<>();
                extra.put("recurrente", true);
                extra.put("patron", patron);
                extra.put("textoRecurrencia", request.getTexto());
                if (node.has("diaSemana") && !node.get("diaSemana").isNull()) {
                    extra.put("diaSemana", node.get("diaSemana").asText());
                }
                if (node.has("diasSemana") && node.get("diasSemana").isArray()) {
                    List<String> dias = new ArrayList<>();
                    node.get("diasSemana").forEach(d -> dias.add(d.asText()));
                    extra.put("diasSemana", dias);
                }
                evento.setDatosExtra(extra);
            }

            eventos.add(evento);
        }

        long elapsed = System.currentTimeMillis() - inicio;

        if (eventos.size() == 1) {
            log.info("✅ IA detectó 1 evento en {}ms", elapsed);
            return new EventoIaResponse(true, null, eventos.get(0), null, elapsed);
        } else {
            log.info("✅ IA detectó {} eventos múltiples en {}ms", eventos.size(), elapsed);
            return new EventoIaResponse(true, null, null, eventos, elapsed);
        }
    }

    private void normalizarHora(ObjectNode node, String campo) {
        if (!node.has(campo) || node.get(campo).isNull()) return;
        String val = node.get(campo).asText("").trim()
                         .replaceAll("(?i)\\s*[Hh].*$", "").trim();
        if (val.isEmpty() || val.equals("null")) { node.putNull(campo); return; }
        if (val.matches("\\d{1,2}:\\d{2}$")) val = val + ":00";
        if (val.matches("\\d{1,2}$"))        val = val + ":00:00";
        node.put(campo, val);
    }

    private void normalizarPersonas(ObjectNode node) {
        if (!node.has("personasRelacionadas")) return;
        JsonNode pr = node.get("personasRelacionadas");
        if (pr.isArray()) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < pr.size(); i++) {
                if (i > 0) sb.append(", ");
                sb.append(pr.get(i).asText());
            }
            String unido = sb.toString().trim();
            if (unido.isEmpty()) node.putNull("personasRelacionadas");
            else                 node.put("personasRelacionadas", unido);
        } else if (pr.isNull() || pr.asText().isBlank()) {
            node.putNull("personasRelacionadas");
        }
    }

        String prompt = contextBuilder.buildPrompt(pregunta, "calendario");
        return ollamaClient.generate(prompt, 1536);

            - "cada día de julio" → "recurrente":true, "patron":"DAILY", fechaInicio:"2026-07-01", fechaFin:"2026-07-31"
            - "primer lunes de cada mes" → "recurrente":true, "patron":"MONTHLY", "diaSemana":"MONDAY", "posicion":"primero"
            - Si NO es recurrente → NO incluyas estos campos

            FECHA CONTEXTO: %s
            TEXTO: "%s"

    }

    private String extraerJson(String respuesta) {
        if (respuesta == null) return "{}";
        respuesta = respuesta.trim();
        int objInicio = respuesta.indexOf('{');
        int objFin    = respuesta.lastIndexOf('}');
        if (objInicio >= 0 && objFin > objInicio) {
            return respuesta.substring(objInicio, objFin + 1);
        }
        return respuesta;
    }

    public Optional<EventoParseResult> parse(String texto, LocalDate fechaContexto) {
        if (texto == null || texto.isBlank()) return Optional.empty();
        String textoNorm = texto.trim().toLowerCase();
        log.debug("🔍 Parseando: {}", textoNorm);

        Optional<EventoParseResult> resultado;

        // ── PRIORIDAD 1: Recurrencia Semanal ──────────────────
        if ((resultado = parseRecSemanalMultipleMes(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecSemanalMes(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecSemanalMultiple(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecSemanalSimple(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecDiariaMes(textoNorm, fechaContexto)).isPresent()) return resultado;

        // ── PRIORIDAD 2: Recurrencia Mensual ──────────────────
        if ((resultado = parseRecMensualRelativo(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecMensualLista(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseRecMensualSimple(textoNorm, fechaContexto)).isPresent()) return resultado;

        // ── PRIORIDAD 3: Recurrencia Anual ────────────────────
        if ((resultado = parseRecAnual(textoNorm, fechaContexto)).isPresent()) return resultado;

        // ── PRIORIDAD 4: Recurrencia con Rango ────────────────
        if ((resultado = parseRecRangoMeses(textoNorm, fechaContexto)).isPresent()) return resultado;

        // ── PRIORIDAD 5: Eventos Únicos ───────────────────────
        if ((resultado = parseCumpleanos(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseCitaMedica(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseTutoria(textoNorm, fechaContexto)).isPresent()) return resultado;
        if ((resultado = parseClaustro(textoNorm, fechaContexto)).isPresent()) return resultado;

        log.debug("⏩ Ningún patrón REGEX coincidió");
        return Optional.empty();
    }

    private Optional<EventoParseResult> parseRecSemanalMultipleMes(String texto, LocalDate hoy) {
        Matcher m = REC_SEMANAL_MULTIPLE_MES.matcher(texto);
        if (!m.find()) return Optional.empty();

        String dia1 = normalizarDia(m.group(2));
        String dia2 = normalizarDia(m.group(3));
        String mes = m.group(4).toLowerCase();

        DayOfWeek dow1 = DIAS_SEMANA.get(dia1);
        DayOfWeek dow2 = DIAS_SEMANA.get(dia2);
        Integer mesNum = MESES.get(mes);

        if (dow1 == null || dow2 == null || mesNum == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente semanal múltiple ({} y {}) de {}", dia1, dia2, mes);
        return crearRecurrenteSemanal(texto, hoy, List.of(dow1, dow2), mesNum);
    }

    private Optional<EventoParseResult> parseRecSemanalMes(String texto, LocalDate hoy) {
        Matcher m = REC_SEMANAL_MES.matcher(texto);
        if (!m.find()) return Optional.empty();

        String dia = normalizarDia(m.group(2));
        String mes = m.group(3).toLowerCase();

        DayOfWeek dow = DIAS_SEMANA.get(dia);
        Integer mesNum = MESES.get(mes);

        if (dow == null || mesNum == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente semanal ({}) de {}", dia, mes);
        return crearRecurrenteSemanal(texto, hoy, List.of(dow), mesNum);
    }

    private Optional<EventoParseResult> parseRecSemanalMultiple(String texto, LocalDate hoy) {
        Matcher m = REC_SEMANAL_MULTIPLE.matcher(texto);
        if (!m.find()) return Optional.empty();

        String dia1 = normalizarDia(m.group(2));
        String dia2 = normalizarDia(m.group(3));

        DayOfWeek dow1 = DIAS_SEMANA.get(dia1);
        DayOfWeek dow2 = DIAS_SEMANA.get(dia2);

        if (dow1 == null || dow2 == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente semanal múltiple ({} y {})", dia1, dia2);
        return crearRecurrenteSemanal(texto, hoy, List.of(dow1, dow2), null);
    }

    private Optional<EventoParseResult> parseRecSemanalSimple(String texto, LocalDate hoy) {
        Matcher m = REC_SEMANAL_SIMPLE.matcher(texto);
        if (!m.find()) return Optional.empty();

        String dia = normalizarDia(m.group(2));
        DayOfWeek dow = DIAS_SEMANA.get(dia);

        if (dow == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente semanal ({})", dia);
        return crearRecurrenteSemanal(texto, hoy, List.of(dow), null);
    }

    private Optional<EventoParseResult> parseRecDiariaMes(String texto, LocalDate hoy) {
        Matcher m = REC_DIARIA_MES.matcher(texto);
        if (!m.find()) return Optional.empty();

        String mes = m.group(2).toLowerCase();
        Integer mesNum = MESES.get(mes);

        if (mesNum == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente diaria de {}", mes);
        return crearRecurrenteDiaria(texto, hoy, mesNum);
    }

    private Optional<EventoParseResult> parseRecMensualSimple(String texto, LocalDate hoy) {
        Matcher m = REC_MENSUAL_SIMPLE.matcher(texto);
        if (!m.find()) return Optional.empty();

        int dia = Integer.parseInt(m.group(1));

        log.info("✅ REGEX: Recurrente mensual (día {})", dia);
        return crearRecurrenteMensual(texto, hoy, dia, null);
    }

    private Optional<EventoParseResult> parseRecMensualLista(String texto, LocalDate hoy) {
        Matcher m = REC_MENSUAL_LISTA.matcher(texto);
        if (!m.find()) return Optional.empty();

        int dia = Integer.parseInt(m.group(1));
        String mesesStr = m.group(2).toLowerCase();

        // Extraer meses: "enero, marzo y mayo" → [1, 3, 5]
        List<Integer> meses = Arrays.stream(mesesStr.split("[,\\sy]+"))
            .map(String::trim)
            .filter(s -> !s.isEmpty() && !s.equals("y"))
            .map(MESES::get)
            .filter(Objects::nonNull)
            .collect(Collectors.toList());

        if (meses.isEmpty()) return Optional.empty();

        log.info("✅ REGEX: Recurrente mensual (día {} de meses {})", dia, meses);
        return crearRecurrenteMensual(texto, hoy, dia, meses);
    }

    private Optional<EventoParseResult> parseRecMensualRelativo(String texto, LocalDate hoy) {
        Matcher m = REC_MENSUAL_RELATIVO.matcher(texto);
        if (!m.find()) return Optional.empty();

        String posStr = m.group(1).toLowerCase();
        String diaStr = m.group(2).toLowerCase();
        String mesStr = m.group(3) != null ? m.group(3).toLowerCase() : null;

        Integer posicion = POSICIONES.get(posStr);
        DayOfWeek dow = DIAS_SEMANA.get(diaStr);
        Integer mes = mesStr != null ? MESES.get(mesStr) : null;

        if (posicion == null || dow == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente mensual relativo ({} {} de {})", 
                 posStr, diaStr, mes != null ? mesStr : "cada mes");
        return crearRecurrenteMensualRelativo(texto, hoy, posicion, dow, mes);
    }

    private Optional<EventoParseResult> parseRecAnual(String texto, LocalDate hoy) {
        Matcher m = REC_ANUAL.matcher(texto);
        if (!m.find()) return Optional.empty();

        int dia = Integer.parseInt(m.group(1));
        String mes = m.group(2).toLowerCase();
        Integer mesNum = MESES.get(mes);

        if (mesNum == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente anual ({} de {})", dia, mes);
        return crearRecurrenteAnual(texto, hoy, dia, mesNum);
    }

    //  PARSERS DE RECURRENCIA CON RANGO
    // ══════════════════════════════════════════════════════════

    private Optional<EventoParseResult> parseRecRangoMeses(String texto, LocalDate hoy) {
        Matcher m = REC_RANGO_MESES.matcher(texto);
        if (!m.find()) return Optional.empty();

        String diaStr = normalizarDia(m.group(1));
        String mesInicio = m.group(2).toLowerCase();
        String mesFin = m.group(3).toLowerCase();

        Integer mesInicioNum = MESES.get(mesInicio);
        Integer mesFinNum = MESES.get(mesFin);

        if (mesInicioNum == null || mesFinNum == null) return Optional.empty();

        boolean esDiario = diaStr.equals("día") || diaStr.equals("dia");
        DayOfWeek dow = esDiario ? null : DIAS_SEMANA.get(diaStr);

        if (!esDiario && dow == null) return Optional.empty();

        log.info("✅ REGEX: Recurrente {} desde {} hasta {}", 
                 esDiario ? "diaria" : diaStr, mesInicio, mesFin);
        return crearRecurrenteRango(texto, hoy, dow, mesInicioNum, mesFinNum);
    }

    private Optional<EventoParseResult> crearRecurrenteSemanal(String texto, LocalDate hoy, 
                                                                 List<DayOfWeek> dias, Integer mes) {
        LocalTime hora = extraerHora(texto);
        String lugar = extraerLugar(texto);
        String persona = extraerPersona(texto);

        LocalDate inicio, fin;
        if (mes != null) {
            YearMonth ym = YearMonth.of(hoy.getYear(), mes);
            if (ym.isBefore(YearMonth.from(hoy))) ym = ym.plusYears(1);
            inicio = ym.atDay(1);
            fin = ym.atEndOfMonth();
        } else {
            inicio = hoy;
            fin = hoy.plusMonths(3);
        }

        Evento.TipoEvento tipo = inferirTipo(texto);
        String titulo = inferirTitulo(texto, tipo, persona);

        Evento plantilla = new Evento();
        plantilla.setTipo(tipo);
        plantilla.setTitulo(titulo);
        plantilla.setFechaInicio(inicio);
        plantilla.setFechaFin(fin);
        plantilla.setHoraInicio(hora);
        plantilla.setLugar(lugar);
        plantilla.setPersonasRelacionadas(persona);
        plantilla.setTextoOriginalVoz(texto);

        Map<String, Object> datosExtra = new HashMap<>();
        datosExtra.put("recurrente", true);
        datosExtra.put("patron", "WEEKLY");
        datosExtra.put("diasSemana", dias.stream().map(DayOfWeek::name).toList());
        plantilla.setDatosExtra(datosExtra);

        return Optional.of(new EventoParseResult(plantilla, true));
    }

    private Optional<EventoParseResult> crearRecurrenteDiaria(String texto, LocalDate hoy, Integer mes) {
        LocalTime hora = extraerHora(texto);
        String lugar = extraerLugar(texto);

    private String normalizarDia(String dia) {
        if (dia == null) return null;
        dia = dia.toLowerCase().trim();
        
        Set<String> diasSingulares = Set.of("lunes", "martes", "miércoles", "miercoles", "jueves", "viernes");
        
        if (diasSingulares.contains(dia)) {
            return dia;
        }
        
        if (dia.endsWith("s")) {
            return dia.substring(0, dia.length() - 1);
        }
        
        return dia;
    }

    private LocalTime extraerHora(String texto) {
        Pattern p = Pattern.compile("(?i)a\\s+las\\s+(\\d{1,2})(?::(\\d{2}))?");
        Matcher m = p.matcher(texto);
        if (m.find()) {
            int hora = Integer.parseInt(m.group(1));
            int min = m.group(2) != null ? Integer.parseInt(m.group(2)) : 0;
            return LocalTime.of(hora, min);
        }
        return null;
    }

    private String extraerLugar(String texto) {
        Pattern p = Pattern.compile("(?i)(?:en|de)\\s+([A-ZÁÉÍÓÚÑ][\\wáéíóúñ]+)(?:[,\\s]|$)");
        Matcher m = p.matcher(texto);
        if (m.find()) {
            String lugar = m.group(1);
            if (!lugar.matches("(?i)(todos|cada|lunes|martes|miércoles|jueves|viernes|sábado|domingo|enero|febrero|marzo|abril|mayo|junio|julio|agosto|septiembre|octubre|noviembre|diciembre)")) {
                return capitalize(lugar);
            }
        }
        return null;
    }

    private String extraerPersona(String texto) {
        Pattern p = Pattern.compile("(?i)(?:de|para)\\s+([A-ZÁÉÍÓÚÑ][\\wáéíóúñ]+)(?:[,\\s]|$)");
        Matcher m = p.matcher(texto);
        if (m.find()) {
            String persona = m.group(1);
            if (!persona.matches("(?i)(todos|cada|lunes|martes|miércoles|jueves|viernes|sábado|domingo|enero|febrero|marzo|abril|mayo|junio|julio|agosto|septiembre|octubre|noviembre|diciembre)")) {
                return capitalize(persona);
            }
        }
        return null;
    }

    private Evento.TipoEvento inferirTipo(String texto) {
        texto = texto.toLowerCase();
        if (texto.contains("trabajo") || texto.contains("presencial") || texto.contains("claustro") || texto.contains("reunión"))
            return Evento.TipoEvento.TRABAJO;
        if (texto.contains("médic") || texto.contains("medic") || texto.contains("doctor"))
            return Evento.TipoEvento.SALUD;
        if (texto.contains("tutoría") || texto.contains("tutoria") || texto.contains("colegio"))
            return Evento.TipoEvento.FAMILIA;
        if (texto.contains("curso") || texto.contains("clase"))
            return Evento.TipoEvento.APRENDIZAJE;
        if (texto.contains("gimnasio") || texto.contains("deporte"))
            return Evento.TipoEvento.OCIO;
        return Evento.TipoEvento.PERSONAL;
    }

    private String inferirTitulo(String texto, Evento.TipoEvento tipo, String persona) {
        if (texto.contains("trabajo presencial") || texto.contains("presencial")) {
            return "Trabajo presencial" + (persona != null ? " de " + persona : "");
        }
        String[] palabras = texto.split("\\s+");
        for (String p : palabras) {
            if (p.length() > 4 && !p.matches("(?i)(todos|cada|lunes|martes|miércoles|jueves|viernes|sábado|domingo|enero|febrero|marzo|abril|mayo|junio|julio|agosto|septiembre|octubre|noviembre|diciembre)")) {
                return capitalize(p) + (persona != null ? " de " + persona : "");
            }
        }
        return tipo.getLabel();
    }

    private String capitalize(String str) {
        if (str == null || str.isEmpty()) return str;
        return str.substring(0, 1).toUpperCase() + str.substring(1).toLowerCase();
    }

}

package com.miempresa.fasi.calendario;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.time.LocalDate;
import java.util.List;
import java.util.UUID;

@Repository
public interface EventoRepository extends JpaRepository<Evento, UUID> {

    // ── Consultas por rango de fechas (para la vista mensual/semanal) ──
    // Incluye eventos que EMPIEZAN en el mes O cuyo rango SOLAPA con el mes
    @Query("""
        SELECT e FROM Evento e
        WHERE e.fechaInicio <= :hasta
          AND (e.fechaFin IS NULL OR e.fechaFin >= :desde)
        ORDER BY e.fechaInicio ASC, e.horaInicio ASC
        """)
    List<Evento> findByMesConRango(@Param("desde") LocalDate desde,
                                   @Param("hasta") LocalDate hasta);

    // Mantener el original para compatibilidad
    List<Evento> findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(
            LocalDate desde, LocalDate hasta);

    // ── Consultas por tipo ──────────────────────────────────
    List<Evento> findByTipoOrderByFechaInicioAsc(Evento.TipoEvento tipo);

    // ── Búsqueda full-text con PostgreSQL tsvector ──────────
    @Query(value = """
        SELECT * FROM eventos
        WHERE texto_busqueda @@ plainto_tsquery('spanish', unaccent(:query))
        ORDER BY fecha_inicio DESC
        LIMIT 50
        """, nativeQuery = true)
    List<Evento> buscarFullText(@Param("query") String query);

    // ── Próximos eventos (para widget de dashboard) ─────────
    @Query("SELECT e FROM Evento e WHERE e.fechaInicio >= :hoy ORDER BY e.fechaInicio ASC, e.horaInicio ASC")
    List<Evento> findProximos(@Param("hoy") LocalDate hoy);

    // ── Eventos de un día concreto ──────────────────────────
    List<Evento> findByFechaInicioOrderByHoraInicioAsc(LocalDate fecha);

    // Eventos con datos pendientes (sin hora, sin lugar, etc.)
    @Query(value = "SELECT * FROM eventos WHERE datos_extra @> '{\"pendienteHora\": true}'::jsonb ORDER BY fecha_inicio ASC", nativeQuery = true)
    List<Evento> findEventosPendientesHora();

    @Query(value = "SELECT * FROM eventos WHERE datos_extra @> '{\"pendienteHora\": true}'::jsonb OR datos_extra @> '{\"pendienteLugar\": true}'::jsonb ORDER BY fecha_inicio ASC", nativeQuery = true)
    List<Evento> findEventosPendientesDatos();

    // ── Consultas optimizadas con JOIN FETCH para evitar N+1 ──
    @Query("SELECT DISTINCT e FROM Evento e LEFT JOIN FETCH e.etiquetas " +
           "WHERE e.fechaInicio <= :fin AND (e.fechaFin IS NULL OR e.fechaFin >= :inicio) " +
           "ORDER BY e.fechaInicio, e.horaInicio")
    List<Evento> findByRangoOptimizado(@Param("inicio") LocalDate inicio, 
                                        @Param("fin") LocalDate fin);

    @Query("SELECT DISTINCT e FROM Evento e LEFT JOIN FETCH e.etiquetas " +
           "WHERE e.fechaInicio >= :desde " +
           "ORDER BY e.fechaInicio, e.horaInicio")
    List<Evento> findProximosOptimizado(@Param("desde") LocalDate desde);

    @Query("SELECT DISTINCT e FROM Evento e LEFT JOIN FETCH e.etiquetas " +
           "ORDER BY e.fechaInicio, e.horaInicio")
    List<Evento> findAllOptimizado();
}

        String patron = datosExtra.get("patron").toString().toUpperCase();
        LocalDate inicio = plantilla.getFechaInicio();
        LocalDate fin = plantilla.getFechaFin() != null ? plantilla.getFechaFin() : inicio.plusMonths(3);

        List<LocalDate> fechas = new ArrayList<>();
        
        switch (patron) {
            case "WEEKLY" -> {
                List<DayOfWeek> dias = extraerDiasSemana(datosExtra);
                LocalDate cursor = inicio.withDayOfMonth(1); // Empezar desde el día 1 del mes
                while (!cursor.isAfter(fin) && fechas.size() < 100) {
                    if (dias.contains(cursor.getDayOfWeek())) {
                        fechas.add(cursor);
                    }
                    cursor = cursor.plusDays(1);
                }
            }
            case "DAILY" -> {
                LocalDate cursor = inicio;
                while (!cursor.isAfter(fin) && fechas.size() < 100) {
                    fechas.add(cursor);
                    cursor = cursor.plusDays(1);
                }
            }
            case "MONTHLY" -> {
                int diaDelMes = (int) datosExtra.getOrDefault("diaDelMes", inicio.getDayOfMonth());
                LocalDate cursor = inicio;
                while (!cursor.isAfter(fin) && fechas.size() < 100) {
                    int diaAjustado = Math.min(diaDelMes, cursor.lengthOfMonth());
                    fechas.add(cursor.withDayOfMonth(diaAjustado));
                    cursor = cursor.plusMonths(1);
                }
            }
            case "MONTHLY_RELATIVE" -> {
                String diaSemana = datosExtra.get("diaSemana").toString();
                int posicion = (int) datosExtra.get("posicion");
                DayOfWeek dow = parseDiaSemana(diaSemana);
                LocalDate cursor = inicio;
                while (!cursor.isAfter(fin) && fechas.size() < 100) {
                    LocalDate fecha = calcularDiaRelativo(cursor, dow, posicion);
                    if (fecha != null && !fecha.isAfter(fin)) {
                        fechas.add(fecha);
                    }
                    cursor = cursor.plusMonths(1);
                }
            }
            case "YEARLY" -> {
                int diaDelMes = (int) datosExtra.get("diaDelMes");
                int mes = (int) datosExtra.get("mes");
                LocalDate cursor = LocalDate.of(inicio.getYear(), mes, diaDelMes);
                if (cursor.isBefore(inicio)) cursor = cursor.plusYears(1);
                while (!cursor.isAfter(fin) && fechas.size() < 10) {
                    fechas.add(cursor);
                    cursor = cursor.plusYears(1);
                }
            }
            default -> fechas.add(inicio);
        }

        log.info("📅 Vista previa: {} eventos generados para patrón {}", fechas.size(), patron);
        return fechas;
    }

    /**
     * Guarda eventos recurrentes en la BD.
     */
    @Transactional
    public List<Evento> saveRecurrente(EventoConfirmarRequest dto) {
        Evento base = dto.toEvento();
        List<LocalDate> fechas = generarVistaPrevia(base);
        
        List<Evento> generados = new ArrayList<>();
        for (LocalDate fecha : fechas) {
            Evento e = new Evento();
            e.setTipo(base.getTipo());
            e.setTitulo(base.getTitulo());
            e.setFechaInicio(fecha);
            e.setFechaFin(null); // Cada evento individual es de 1 día
            e.setHoraInicio(base.getHoraInicio());
            e.setHoraFin(base.getHoraFin());
            e.setTodoDia(base.isTodoDia());
            e.setLugar(base.getLugar());
            e.setPersonasRelacionadas(base.getPersonasRelacionadas());
            e.setDescripcion(base.getDescripcion());
            e.setRecordatorioMinutos(base.getRecordatorioMinutos());
            e.setDatosExtra(base.getDatosExtra());
            e.setEtiquetas(base.getEtiquetas());
            e.setTextoOriginalVoz(base.getTextoOriginalVoz());
            generados.add(repo.save(e));
        }

        log.info("✅ Guardados {} eventos recurrentes", generados.size());
        return generados;
    }

    //  UTILIDADES PRIVADAS
    // ══════════════════════════════════════════════════════════

    private List<DayOfWeek> extraerDiasSemana(Map<String, Object> datosExtra) {
        List<DayOfWeek> dias = new ArrayList<>();
        
        // Intentar leer "diasSemana" (plural, lista)
        if (datosExtra.containsKey("diasSemana")) {
            Object obj = datosExtra.get("diasSemana");
            if (obj instanceof List<?> lista) {
                for (Object item : lista) {
                    DayOfWeek dow = parseDiaSemana(item.toString());
                    if (dow != null) dias.add(dow);
                }
            }
        }
        
        // Fallback: leer "diaSemana" (singular, string)
        if (dias.isEmpty() && datosExtra.containsKey("diaSemana")) {
            DayOfWeek dow = parseDiaSemana(datosExtra.get("diaSemana").toString());
            if (dow != null) dias.add(dow);
        }
        
        return dias;
    }

    private DayOfWeek parseDiaSemana(String nombre) {
        return switch (nombre.toUpperCase().trim()) {
            case "LUNES",     "MONDAY"    -> DayOfWeek.MONDAY;
            case "MARTES",    "TUESDAY"   -> DayOfWeek.TUESDAY;
            case "MIERCOLES", "MIÉRCOLES", "WEDNESDAY" -> DayOfWeek.WEDNESDAY;
            case "JUEVES",    "THURSDAY"  -> DayOfWeek.THURSDAY;
            case "VIERNES",   "FRIDAY"    -> DayOfWeek.FRIDAY;
            case "SABADO",    "SÁBADO",   "SATURDAY"  -> DayOfWeek.SATURDAY;
            case "DOMINGO",   "SUNDAY"    -> DayOfWeek.SUNDAY;
            default -> null;
        };
    }

    private LocalDate calcularDiaRelativo(LocalDate mes, DayOfWeek dow, int posicion) {
        if (posicion == -1) {
            // Último día de la semana del mes
            return mes.with(TemporalAdjusters.lastInMonth(dow));
        } else {
            // Primer, segundo, tercer, cuarto día de la semana
            return mes.with(TemporalAdjusters.dayOfWeekInMonth(posicion, dow));
        }
    }

    //  MÉTODO LEGACY (mantener por compatibilidad)
    // ══════════════════════════════════════════════════════════

    @Transactional
    public void saveRecurrente(Evento plantilla, String patron) {
        LocalDate inicio = plantilla.getFechaInicio();
        LocalDate fin = plantilla.getFechaFin() != null ? plantilla.getFechaFin() : inicio.plusMonths(3);
        log.info("🔄 Generando eventos recurrentes: {} desde {} hasta {}", patron, inicio, fin);
        LocalDate actual = inicio;
        int count = 0;
        while (!actual.isAfter(fin) && count < 100) {
            Evento copia = new Evento();
            copia.setTipo(plantilla.getTipo());
            copia.setTitulo(plantilla.getTitulo());
            copia.setFechaInicio(actual);
            copia.setFechaFin(actual);
            copia.setHoraInicio(plantilla.getHoraInicio());
            copia.setHoraFin(plantilla.getHoraFin());
            copia.setTodoDia(plantilla.isTodoDia());
            copia.setLugar(plantilla.getLugar());
            copia.setPersonasRelacionadas(plantilla.getPersonasRelacionadas());
            copia.setDescripcion(plantilla.getDescripcion());
            copia.setRecordatorioMinutos(plantilla.getRecordatorioMinutos());
            copia.setEtiquetas(plantilla.getEtiquetas());
            repo.save(copia);
            count++;
            switch (patron.toUpperCase()) {
                case "DAILY": actual = actual.plusDays(1); break;
                case "WEEKLY": actual = actual.plusWeeks(1); break;
                case "MONTHLY": actual = actual.plusMonths(1); break;
                default: log.warn("Patrón desconocido: {}", patron); return;
            }
        }
        log.info("✅ Creados {} eventos recurrentes", count);
    }

    @ResponseBody
    public void noResource() { }

    @ResponseStatus(HttpStatus.NOT_FOUND)

        return "error/404";
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    @ResponseBody
    public ApiResponse<Void> validationError(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + ": " + e.getDefaultMessage())
            .collect(Collectors.joining(", "));
        return ApiResponse.error(msg);
    }

    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public String genericError(Exception ex, Model model) {
        log.error("Error no controlado", ex);
        model.addAttribute("error", "Error interno del servidor");
        return "error/500";
    }

package com.miempresa.fasi.contexto;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.time.*;
import java.time.format.DateTimeFormatter;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Optional;

/**
 * Servicio de cálculo astronómico local para FASI.
 *
 * MAREAS: algoritmo semidiurno basado en pleamar de referencia.
 *   Ciclo: 12h 25min entre pleamares (ciclo lunar M2).
 *   Precisión: ±20-30 min respecto a tablas oficiales.
 *
 * SOL: fórmula NOAA simplificada con latitud/longitud reales.
 *   Precisión: ±2 min.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class MareasSolarService {

    private final ContextoFamiliarRepository repo;

    // Ciclo M2 en minutos (12h 25min 14s)
    private static final double CICLO_MAREA_MIN = 12.0 * 60.0 + 25.0 + 14.0 / 60.0;

    // ── API pública ───────────────────────────────────────────

    /**
     * Devuelve todos los datos contextuales para una fecha dada.
     * Formato listo para el footer del calendario.
     */
    public Map<String, String> getDatosContextuales(LocalDate fecha) {
        Map<String, String> datos = new LinkedHashMap<>();

        try {
            double lat = getDouble("ubicacion_lat", 36.7749);
            double lon = getDouble("ubicacion_lon", -6.3536);
            ZoneId zona = ZoneId.of(getString("ubicacion_zona", "Europe/Madrid"));

            // ── Sol ──────────────────────────────────────────
            LocalTime[] sol = calcularSol(fecha, lat, lon, zona);
            datos.put("amanecer",    formatHora(sol[0]));
            datos.put("atardecer",   formatHora(sol[1]));

            // ── Mareas ───────────────────────────────────────
            LocalDateTime refTs = LocalDateTime.parse(
                getString("marea_referencia_ts", "2026-05-05T17:00:00"));
            double refAltura = getDouble("marea_referencia_h", 3.45);

            MareasDia mareas = calcularMareasDia(fecha, refTs, refAltura, zona);
            datos.put("pleamar1",    mareas.pleamar1 != null ? formatHora(mareas.pleamar1.toLocalTime()) : "--:--");
            datos.put("bajamar1",    mareas.bajamar1 != null ? formatHora(mareas.bajamar1.toLocalTime()) : "--:--");
            datos.put("pleamar2",    mareas.pleamar2 != null ? formatHora(mareas.pleamar2.toLocalTime()) : "--:--");
            datos.put("bajamar2",    mareas.bajamar2 != null ? formatHora(mareas.bajamar2.toLocalTime()) : "--:--");
            datos.put("altura_pm",   String.format("%.1f m", refAltura));

            // ── Ciudad ───────────────────────────────────────
            datos.put("ciudad",      getString("ubicacion_ciudad", "Sanlúcar de Barrameda"));

        } catch (Exception e) {
            log.warn("Error calculando datos contextuales: {}", e.getMessage());
            datos.put("amanecer",  "--:--");
            datos.put("atardecer", "--:--");
            datos.put("pleamar1",  "--:--");
            datos.put("bajamar1",  "--:--");
            datos.put("pleamar2",  "--:--");
            datos.put("bajamar2",  "--:--");
            datos.put("ciudad",    "Sanlúcar de Barrameda");
        }

        return datos;
    }

    // ── Cálculo solar (NOAA simplificado) ────────────────────

    private LocalTime[] calcularSol(LocalDate fecha, double lat, double lon, ZoneId zona) {
        // Día del año
        int N = fecha.getDayOfYear();

        // Declinación solar
        double decl = Math.toRadians(23.45 * Math.sin(Math.toRadians(360.0 / 365.0 * (N - 81))));

        // Ecuación del tiempo (minutos)
        double B = Math.toRadians(360.0 / 365.0 * (N - 81));
        double ET = 9.87 * Math.sin(2 * B) - 7.53 * Math.cos(B) - 1.5 * Math.sin(B);

        // Corrección de longitud (minutos)
        double offsetUTC = zona.getRules().getOffset(fecha.atStartOfDay()).getTotalSeconds() / 3600.0;
        double corrLon   = (lon - (offsetUTC * 15.0)) * 4.0; // 4 min por grado

        // Hora solar del mediodía
        double mediodiaSolar = 720.0 - corrLon - ET; // minutos desde medianoche

        // Ángulo horario del amanecer/atardecer
        double latRad = Math.toRadians(lat);
        double cosH   = -Math.tan(latRad) * Math.tan(decl);
        cosH = Math.max(-1.0, Math.min(1.0, cosH)); // clamp
        double H = Math.toDegrees(Math.acos(cosH)) * 4.0; // minutos

        LocalTime amanecer  = minutosAHora((int)(mediodiaSolar - H));
        LocalTime atardecer = minutosAHora((int)(mediodiaSolar + H));

        return new LocalTime[]{ amanecer, atardecer };
    }

    // ── Cálculo de mareas ────────────────────────────────────

    private MareasDia calcularMareasDia(LocalDate fecha, LocalDateTime refPleamar,
                                         double alturaRef, ZoneId zona) {
        // Minutos desde la pleamar de referencia hasta medianoche del día pedido
        LocalDateTime medianoche = fecha.atStartOfDay();
        long minDesdeRef = Duration.between(refPleamar, medianoche).toMinutes();

        // Encontrar la primera pleamar del día (avanzar en ciclos de CICLO_MAREA_MIN)
        double ciclos = minDesdeRef / CICLO_MAREA_MIN;
        long cicloBase = (long) Math.floor(ciclos);

        MareasDia resultado = new MareasDia();
        int encontradas = 0;

        // Buscar hasta 4 eventos de marea en el día (2 pleamares + 2 bajamares)
        for (long i = cicloBase - 1; i <= cicloBase + 4 && encontradas < 4; i++) {
            // Pleamar
            double minPleamar = refPleamar.atZone(zona).toEpochSecond() / 60.0
                              + i * CICLO_MAREA_MIN;
            LocalDateTime tsPleamar = LocalDateTime.ofEpochSecond(
                (long)(minPleamar * 60), 0, zona.getRules().getOffset(medianoche));

            if (tsPleamar.toLocalDate().equals(fecha)) {
                if (resultado.pleamar1 == null) {
                    resultado.pleamar1 = tsPleamar;
                } else if (resultado.pleamar2 == null) {
                    resultado.pleamar2 = tsPleamar;
                }
                encontradas++;
            }

            // Bajamar (media marea después de la pleamar)
            double minBajamar = minPleamar + CICLO_MAREA_MIN / 2.0;
            LocalDateTime tsBajamar = LocalDateTime.ofEpochSecond(
                (long)(minBajamar * 60), 0, zona.getRules().getOffset(medianoche));

            if (tsBajamar.toLocalDate().equals(fecha)) {
                if (resultado.bajamar1 == null) {
                    resultado.bajamar1 = tsBajamar;
                } else if (resultado.bajamar2 == null) {
                    resultado.bajamar2 = tsBajamar;
                }
                encontradas++;
            }
        }

        return resultado;
    }

    // ── Helpers ──────────────────────────────────────────────

    private String getString(String clave, String defecto) {

            return Double.parseDouble(getString(clave, String.valueOf(defecto)));
        } catch (NumberFormatException e) {

        }
    }

    private LocalTime minutosAHora(int minutos) {
        minutos = Math.max(0, Math.min(1439, minutos));
        return LocalTime.of(minutos / 60, minutos % 60);

    }

package com.miempresa.fasi.ia;

import com.miempresa.fasi.calendario.Evento;
import com.miempresa.fasi.calendario.EventoRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.util.List;
import java.util.stream.Collectors;

@Component
@RequiredArgsConstructor
public class IaContextBuilder {

    private final EventoRepository eventoRepo;

    private static final int MAX_EVENTOS_CTX   = 15;
    private static final int RANGO_DIAS_BASE   = 30;
    private static final int RANGO_DIAS_AMPLIO = 90;

    // ════════════════════════════════════════════════════════
    //  Prompt principal — compacto y adaptativo
    // ════════════════════════════════════════════════════════
    public String buildPrompt(String preguntaUsuario, String modulo) {
        String ctx = buildContextoFiltrado(preguntaUsuario);

        return """
            Eres FASI, asistente del calendario. Responde en español, breve y claro.
            Solo usa los datos del CONTEXTO. Si falta un dato responde:
            "No tengo ese dato registrado."
            Nunca inventes información.

            %s

            Pregunta: %s
            Respuesta:
            """.formatted(ctx, preguntaUsuario);
    }

    // ════════════════════════════════════════════════════════
    //  Contexto FILTRADO según palabras clave de la pregunta
    // ════════════════════════════════════════════════════════
    private String buildContextoFiltrado(String pregunta) {
        String p = pregunta.toLowerCase();
        LocalDate hoy = LocalDate.now();

        Evento.TipoEvento tipoFiltro = detectarTipo(p);
        int dias = detectarRangoDias(p);
        LocalDate fin = hoy.plusDays(dias);

        List<Evento> eventos;
        if (tipoFiltro != null) {
            eventos = eventoRepo.findByTipoOrderByFechaInicioAsc(tipoFiltro).stream()
                    .filter(e -> !e.getFechaInicio().isBefore(hoy)
                              && !e.getFechaInicio().isAfter(fin))
                    .limit(MAX_EVENTOS_CTX)
                    .collect(Collectors.toList());
        } else {
            eventos = eventoRepo
                    .findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(hoy, fin)
                    .stream()
                    .limit(MAX_EVENTOS_CTX)
                    .collect(Collectors.toList());
        }

        StringBuilder ctx = new StringBuilder("=== CONTEXTO ===\n");
        ctx.append("Hoy: ").append(hoy).append("\n");
        if (tipoFiltro != null) {
            ctx.append("Filtro tipo: ").append(tipoFiltro.getLabel()).append("\n");
        }
        ctx.append("Rango: próximos ").append(dias).append(" días\n\n");

        if (eventos.isEmpty()) {
            ctx.append("Sin eventos en el rango consultado.\n");
        } else {
            ctx.append("Eventos:\n");
            eventos.forEach(e -> {
                ctx.append("- ").append(e.getFechaInicio())
                   .append(e.getHoraInicio() != null ? " " + e.getHoraInicio() : "")
                   .append(" | ").append(e.getTipo().getLabel())
                   .append(" | ").append(e.getTitulo())
                   .append(e.getLugar() != null ? " (" + e.getLugar() + ")" : "")
                   .append(e.getPersonasRelacionadas() != null
                           ? " · " + e.getPersonasRelacionadas() : "");
                if (e.getEtiquetas() != null && !e.getEtiquetas().isEmpty()) {
                    ctx.append(" #").append(String.join(" #", e.getEtiquetas()));
                }
                ctx.append("\n");
            });
        }
        return ctx.toString();
    }

    // ════════════════════════════════════════════════════════
    //  Heurísticas — sin IA, en Java puro
    //  Mapeo a las 10 macrocategorías
    // ════════════════════════════════════════════════════════
    private Evento.TipoEvento detectarTipo(String p) {
        // SALUD
        if (p.contains("médic") || p.contains("medic") || p.contains("doctor")
         || p.contains("hospital") || p.contains("consulta") || p.contains("dentist")
         || p.contains("fisio") || p.contains("gym") || p.contains("gimnasio")
         || p.contains("terap"))                                  return Evento.TipoEvento.SALUD;

        // SOCIAL
        if (p.contains("cumple") || p.contains("comida") || p.contains("comer")
         || p.contains("cena") || p.contains("almuerz") || p.contains("fiesta")
         || p.contains("quedad") || p.contains("amig"))           return Evento.TipoEvento.SOCIAL;

        // TRABAJO
        if (p.contains("claustro") || p.contains("reuni")
         || p.contains("trabajo") || p.contains("oficina"))       return Evento.TipoEvento.TRABAJO;

        // APRENDIZAJE
        if (p.contains("curso") || p.contains("formaci") || p.contains("clase")
         || p.contains("estudi") || p.contains("taller")
         || p.contains("charla"))                                 return Evento.TipoEvento.APRENDIZAJE;

        // FAMILIA
        if (p.contains("tutor") || p.contains("extraescolar")
         || p.contains("hijo") || p.contains("hija")
         || p.contains("niñ") || p.contains("famil")
         || p.contains("colegio"))                                return Evento.TipoEvento.FAMILIA;

        // VIAJE
        if (p.contains("viaj") || p.contains("vacacion") || p.contains("vuelo")
         || p.contains("hotel"))                                  return Evento.TipoEvento.VIAJE;

        // GESTION
        if (p.contains("compra") || p.contains("trámit") || p.contains("tramit")
         || p.contains("banco") || p.contains("recad")
         || p.contains("itv"))                                    return Evento.TipoEvento.GESTION;

        // OCIO
        if (p.contains("hobby") || p.contains("deporte") || p.contains("museo")
         || p.contains("cine") || p.contains("teatro")
         || p.contains("concierto"))                              return Evento.TipoEvento.OCIO;

        return null;
    }

    private int detectarRangoDias(String p) {
        if (p.contains("hoy"))                                   return 1;
        if (p.contains("mañana"))                                return 2;
        if (p.contains("esta semana") || p.contains("semana"))   return 7;
        if (p.contains("quincena") || p.contains("15 días"))     return 15;
        if (p.contains("este mes") || p.contains("mes"))         return RANGO_DIAS_BASE;
        if (p.contains("trimestre") || p.contains("3 meses"))    return RANGO_DIAS_AMPLIO;
        if (p.contains("año") || p.contains("anual"))            return 365;
        return RANGO_DIAS_BASE;
    }
}

package com.miempresa.fasi.ia;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Slf4j
@Service
@RequiredArgsConstructor
public class IaService {

    private final OllamaClient       ollamaClient;
    private final IaContextBuilder   contextBuilder;
    private final IaMensajeRepository repo;

    @Transactional
    public IaChatResponse chat(IaChatRequest request) {
        long inicio = System.currentTimeMillis();

        String prompt    = contextBuilder.buildPrompt(request.getMensaje(), request.getModulo());
        String respuesta = ollamaClient.generate(prompt);
        long   tiempoMs  = System.currentTimeMillis() - inicio;

        // Persistir en historial
        repo.save(IaMensaje.builder()
                .pregunta(request.getMensaje())
                .respuesta(respuesta)
                .modulo(request.getModulo() != null ? request.getModulo() : "general")
                .tiempoMs(tiempoMs)
                .build());

        log.info("IA respondió en {}ms para módulo '{}'", tiempoMs, request.getModulo());
        return new IaChatResponse(respuesta, request.getModulo(), tiempoMs);
    }

    public List<IaMensaje> historial() {
        return repo.findTop20ByOrderByCreatedAtDesc();
    }
}

package com.miempresa.fasi.ia;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonInclude;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.Data;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;
import java.util.Map;

@Slf4j
@Component
@RequiredArgsConstructor
public class OllamaClient {

    @Value("${fasi.ollama.url:http://localhost:11434}")
    private String ollamaUrl;

    @Value("${fasi.ollama.model:qwen2.5:3b}")
    private String model;

    private final ObjectMapper objectMapper;
    private final HttpClient   httpClient = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(10))
            .build();

    // ════════════════════════════════════════════════════════
    //  Generación estándar (chat / consulta libre)
    // ════════════════════════════════════════════════════════
    public String generate(String prompt) {
        return generate(prompt, 1024);
    }

    public String generate(String prompt, int numCtx) {
        try {
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt(prompt);
            req.setStream(false);
            req.setKeep_alive(-1);
            req.setOptions(Map.of(
                "num_ctx",     numCtx,
                "temperature", 0,
                "top_k",       20,
                "top_p",       0.7
            ));

            return callOllama(req);

        } catch (Exception e) {
            log.error("Error llamando a Ollama", e);
            return "⚠️ Error al conectar con el asistente IA. " +
                   "Asegúrate de que Ollama está corriendo en WSL (ollama serve).";
        }
    }

    // ════════════════════════════════════════════════════════
    //  Generación JSON-mode (extracción de eventos)
    //  - format:"json" garantiza salida JSON válida
    //  - num_predict limita la salida -> más rápido
    //  - top_k=10 -> menos creatividad, más velocidad
    // ════════════════════════════════════════════════════════
    public String generateJson(String prompt, int numCtx, int maxOutputTokens) {
        try {
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt(prompt);
            req.setStream(false);
            req.setKeep_alive(-1);
            req.setFormat("json");
            req.setOptions(Map.of(
                "num_ctx",     numCtx,
                "num_predict", maxOutputTokens,
                "temperature", 0,
                "top_k",       10,
                "top_p",       0.5
            ));

            return callOllama(req);

        } catch (Exception e) {
            log.error("Error llamando a Ollama (JSON mode)", e);
            return "{}";
        }
    }

    // ════════════════════════════════════════════════════════
    //  Warmup
    // ════════════════════════════════════════════════════════
    public void warmup() {
        try {
            log.info("🔥 Precalentando modelo IA ({})...", model);
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt("");
            req.setStream(false);
            req.setKeep_alive(-1);
            req.setOptions(Map.of("num_ctx", 512, "temperature", 0));

            callOllama(req);
            log.info("✅ Modelo IA listo en RAM.");
        } catch (Exception e) {
            log.warn("⚠️ No se pudo precalentar el modelo: {}", e.getMessage());
        }
    }

    // ════════════════════════════════════════════════════════
    //  HTTP helper
    // ════════════════════════════════════════════════════════
    private String callOllama(OllamaRequest req) throws Exception {
        String body = objectMapper.writeValueAsString(req);

        HttpRequest httpReq = HttpRequest.newBuilder()
                .uri(URI.create(ollamaUrl + "/api/generate"))
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(body))
                .timeout(Duration.ofSeconds(120))
                .build();

        HttpResponse<String> resp = httpClient.send(
                httpReq, HttpResponse.BodyHandlers.ofString());

        OllamaResponse ollamaResp =
                objectMapper.readValue(resp.body(), OllamaResponse.class);
        return ollamaResp.getResponse();
    }

    // ════════════════════════════════════════════════════════
    //  DTOs internos
    // ════════════════════════════════════════════════════════
    @Data
    @JsonInclude(JsonInclude.Include.NON_NULL)
    static class OllamaRequest {
        private String              model;
        private String              prompt;
        private boolean             stream;
        private int                 keep_alive;
        private String              format;     // 👈 nuevo: "json" para JSON mode
        private Map<String, Object> options;
    }

    @Data
    @JsonIgnoreProperties(ignoreUnknown = true)
    static class OllamaResponse {
        private String  response;
        private boolean done;
    }
}

    function iniciarMicrofono() {
      console.log('iniciarMicrofono llamado');
      if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        alert('Tu navegador no soporta reconocimiento de voz. Usa Chrome o Edge.'); return;
      }
      if (location.protocol !== 'https:' && location.hostname !== 'localhost' && location.hostname !== '127.0.0.1') {
        alert('El micrófono requiere HTTPS. Accede via localhost o activa HTTPS.'); return;
      }
      navigator.mediaDevices.getUserMedia({ audio: true })
        .then(function(stream) {
          stream.getTracks().forEach(function(t){ t.stop(); });
          _iniciarReconocimiento();
        })
        .catch(function(err) {
          alert('Permiso de micrófono denegado: ' + err.message);
        });
    }
    function _iniciarReconocimiento() {
      var SR = window.SpeechRecognition || window.webkitSpeechRecognition;
      recognition = new SR();
      recognition.lang           = 'es-ES';
      recognition.continuous     = true;
      recognition.interimResults = true;
      var textoAcumulado = '';
      recognition.onresult = function(event) {
        var interino = '';
        for (var i = event.resultIndex; i < event.results.length; i++) {
          if (event.results[i].isFinal) {
            textoAcumulado += event.results[i][0].transcript + ' ';
          } else {
            interino += event.results[i][0].transcript;
          }
        }
        var textarea = document.getElementById('voz-texto');
        textarea.value = textoAcumulado + interino;
      };
      recognition.onend = function() {
        if (grabando) setTimeout(function() { try { recognition.start(); } catch(e){} }, 300);
      };
      recognition.onerror = function(e) {
        console.error('Speech error:', e.error);
        if (e.error === 'no-speech') {
          grabando = false; actualizarBotonMicro(false);
          return;
        }
        if (e.error === 'not-allowed') { alert('Permiso de micrófono denegado. Revisa la configuración del navegador.'); }
        if (e.error === 'network')     { alert('Error de red en reconocimiento de voz.'); }
        grabando = false; actualizarBotonMicro(false);
      };
      recognition.continuous = false;
      recognition.interimResults = false;
      recognition.lang = 'es-ES';
      recognition.maxAlternatives = 1;
      recognition.start();
      console.log('SpeechRecognition iniciado');
      grabando = true;
      actualizarBotonMicro(true);
    }

    async function confirmarEvento() {

            var r = await fetch(ep, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(ev)
            });
            if (r.ok) guardados++; else errores++;
          } catch(e) { errores++; }
        }
        var msg = guardados + ' evento' + (guardados !== 1 ? 's' : '') + ' creado' + (guardados !== 1 ? 's' : '');
        if (errores > 0) msg += ' (' + errores + ' con error)';
        alert(msg);
        cerrarModalVoz();
        window.location.reload();
        return;
      }

      var esRec    = eventoActual.datosExtra && eventoActual.datosExtra.recurrente === true;
      var endpoint = esRec ? '/calendario/api/confirmar-recurrente' : '/calendario/api/confirmar';
      try {
        var resp = await fetch(endpoint, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(eventoActual)
        });
        if (resp.ok) {

        alert('Error: ' + err.message);
      }

    async function comprobarRecordatorios() {
      if (Notification.permission !== 'granted') return;
      try {
        var ahora  = new Date();
        var parts2 = YEAR_MONTH.split('-');
        var y = parseInt(parts2[0],10);

        var r1 = await fetch('/calendario/api/mes?year=' + y + '&month=' + m);
        var eventos = await r1.json();
        eventos.forEach(function(e) {
          if (!e.recordatorioMinutos || !e.fechaInicio) return;
          var horaStr  = e.horaInicio ? e.horaInicio.substring(0,5) : '00:00';
          var eventoTs = new Date(e.fechaInicio + 'T' + horaStr + ':00');
          if (isNaN(eventoTs.getTime())) return;
          var dispararEn = new Date(eventoTs.getTime() - e.recordatorioMinutos * 60000);
          if (Math.abs(ahora - dispararEn) > 90000) return;
          var clave = e.id + '-' + e.recordatorioMinutos;
          if (DISPARADOS.has(clave)) return;
          DISPARADOS.add(clave);
          var minRest = Math.round((eventoTs - ahora) / 60000);
          var tiempo  = minRest <= 0 ? 'ahora' : 'en ' + minRest + ' min';
          var emoji   = (e.tipo && e.tipo.emoji) ? e.tipo.emoji : '';
          var label   = (e.tipo && e.tipo.label) ? e.tipo.label : '';

<script>
(function() {
  const grid = document.querySelector('#calendario-grid, .calendario-grid, #calendar-grid, main');
  if (!grid) return;

  let touchStartX = 0;
  let touchStartY = 0;
  let touchEndX   = 0;
  let touchEndY   = 0;
  const MIN_SWIPE = 60;
  const MAX_VERT  = 80;

  grid.addEventListener('touchstart', e => {
    touchStartX = e.changedTouches[0].screenX;
    touchStartY = e.changedTouches[0].screenY;
  }, { passive: true });

  grid.addEventListener('touchend', e => {
    touchEndX = e.changedTouches[0].screenX;
    touchEndY = e.changedTouches[0].screenY;
    handleSwipe();
  }, { passive: true });

  function handleSwipe() {
    const dx = touchEndX - touchStartX;
    const dy = Math.abs(touchEndY - touchStartY);
    if (Math.abs(dx) < MIN_SWIPE) return;
    if (dy > MAX_VERT) return;
    if (typeof cambiarMes !== 'function') return;
    if (dx > 0) cambiarMes(-1);
    else        cambiarMes(1);
  }
})();
</script>

    function setModulo(btn, modulo) {
      document.getElementById('modulo-input').value = modulo;
      document.querySelectorAll('.modulo-btn').forEach(b => {
        b.classList.remove('bg-indigo-600','text-white');
        b.classList.add('bg-gray-200','text-gray-700');
      });
      btn.classList.add('bg-indigo-600','text-white');
      btn.classList.remove('bg-gray-200','text-gray-700');
    }

    function setSugerencia(texto) {
      document.getElementById('mensaje-input').value = texto;
      document.getElementById('mensaje-input').focus();
    }

    document.getElementById('mensaje-input').addEventListener('keydown', function(e) {
      if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); enviarMensaje(); }
    });

    async function enviarMensaje() {
      const mensajeInput = document.getElementById('mensaje-input');
      const moduloInput  = document.getElementById('modulo-input');
      const chatArea     = document.getElementById('chat-mensajes');
      const loading      = document.getElementById('ia-loading');
      const btnEnviar    = document.getElementById('btn-enviar');
      const mensaje = mensajeInput.value.trim();
      if (!mensaje) return;

      chatArea.insertAdjacentHTML('beforeend', `
        <div class="flex gap-3 items-start justify-end">
          <div class="bg-gray-100 dark:bg-gray-700 rounded-xl px-4 py-3 max-w-2xl">
            <p class="text-sm text-gray-800 dark:text-gray-100">${escapeHtml(mensaje)}</p>
          </div>
          <span class="text-2xl">👤</span>
        </div>`);

      mensajeInput.value    = '';
      btnEnviar.disabled    = true;
      btnEnviar.textContent = '⏳';
      loading.style.display = 'flex';
      chatArea.scrollTop    = chatArea.scrollHeight;

      try {
        const resp = await fetch('/ia/api/chat', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
          body:    JSON.stringify({ mensaje, modulo: moduloInput.value || 'general' })
        });
        if (!resp.ok) throw new Error('HTTP ' + resp.status + ' — ' + await resp.text());
        const data = await resp.json();
        chatArea.insertAdjacentHTML('beforeend', `
          <div class="flex gap-3 items-start">
            <span class="text-2xl">🤖</span>
            <div class="bg-indigo-50 dark:bg-indigo-900/30 rounded-xl px-4 py-3 max-w-2xl">
              <p class="text-sm text-gray-700 dark:text-gray-200 whitespace-pre-wrap">${escapeHtml(data.respuesta)}</p>
              <p class="text-xs text-gray-400 mt-2">⏱ ${data.tiempoMs}ms · ${data.modulo || 'general'}</p>
            </div>
          </div>`);
      } catch (err) {
        chatArea.insertAdjacentHTML('beforeend', `
          <div class="flex gap-3 items-start">
            <span class="text-2xl">🤖</span>
            <div class="bg-red-50 dark:bg-red-900/30 border border-red-200 rounded-xl px-4 py-3 max-w-2xl">
              <p class="text-sm text-red-600 font-semibold">⚠️ Error al conectar con Gemma</p>
              <p class="text-xs text-red-500 mt-1">${escapeHtml(err.message)}</p>
              <p class="text-xs text-gray-400 mt-2">Verifica en WSL: <code>ollama serve</code></p>
            </div>
          </div>`);
      } finally {
        loading.style.display = 'none';
        btnEnviar.disabled    = false;
        btnEnviar.textContent = 'Enviar →';
        chatArea.scrollTop    = chatArea.scrollHeight;
        mensajeInput.focus();
      }
    }

    function escapeHtml(text) {
      const d = document.createElement('div');
      d.appendChild(document.createTextNode(String(text)));
      return d.innerHTML;
    }

    window.addEventListener('load', () => {
      const c = document.getElementById('chat-mensajes');
      if (c) c.scrollTop = c.scrollHeight;
    });
  </script>

</section>

  <script>
    // ── Dark mode ────────────────────────────────────────────
    const htmlEl  = document.documentElement;

    function applyDark(isDark) {
      isDark ? htmlEl.classList.add('dark') : htmlEl.classList.remove('dark');
      const menuIcon = document.getElementById('menu-dark-icon');
      if (menuIcon) menuIcon.textContent = isDark ? '☀️' : '🌙';
    }
    
    function toggleDark() {
      const isDark = !htmlEl.classList.contains('dark');
      localStorage.setItem('fasi-dark', isDark ? '1' : '0');
      applyDark(isDark);
    }
    
    (function() {
      const saved = localStorage.getItem('fasi-dark');
      applyDark(saved !== null ? saved === '1'
                               : window.matchMedia('(prefers-color-scheme: dark)').matches);
    })();

    // ── Menú hamburguesa ─────────────────────────────────────
    const menuOverlay   = document.getElementById('menu-overlay');
    const hamburgerMenu = document.getElementById('hamburger-menu');

    function toggleMenu() {
      const isHidden = hamburgerMenu.classList.contains('hidden-menu');
      if (isHidden) {
        hamburgerMenu.classList.remove('hidden-menu');
        menuOverlay.classList.remove('hidden');
      } else {
        cerrarMenu();
      }
    }
    
    function cerrarMenu() {
      hamburgerMenu.classList.add('hidden-menu');
      menuOverlay.classList.add('hidden');
    }

    // ── Buscador IA ──────────────────────────────────────────
    function abrirBuscadorIA() {
      document.getElementById('modal-buscador').classList.remove('hidden');
      setTimeout(() => document.getElementById('ia-search-input').focus(), 100);
    }
    
    function cerrarBuscadorIA() {
      document.getElementById('modal-buscador').classList.add('hidden');
    }

    document.getElementById('ia-search-input').addEventListener('keydown', e => {
      if (e.key === 'Enter') buscarIA(e.target.value.trim());
      if (e.key === 'Escape') cerrarBuscadorIA();
    });

    async function buscarIA(pregunta) {
      if (!pregunta) pregunta = document.getElementById('ia-search-input').value.trim();
      if (!pregunta) return;
      document.getElementById('ia-search-input').value = pregunta;

      const result  = document.getElementById('ia-search-result');
      const spinner = document.getElementById('ia-search-spinner');
      result.innerHTML  = '';
      spinner.style.display = 'flex';

      try {
        const resp = await fetch('/calendario/api/consulta', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify(pregunta)
        });
        const texto = await resp.text();
        result.innerHTML = `
          <div class="text-sm text-gray-700 dark:text-gray-200 whitespace-pre-wrap leading-relaxed">
            ${escapeHtml(texto)}
          </div>`;
      } catch (err) {
        result.innerHTML = `<p class="text-sm text-red-500">Error: ${escapeHtml(err.message)}</p>`;
      } finally {
        spinner.style.display = 'none';
      }
    }

-- ============================================================
--  FASI Calendario — V8
--  Tabla principal de eventos con búsqueda full-text
-- ============================================================

-- Extensión para búsqueda full-text en español
CREATE EXTENSION IF NOT EXISTS unaccent;

-- Wrapper inmutable de unaccent (necesario para columnas GENERATED)
CREATE OR REPLACE FUNCTION unaccent_immutable(text)
RETURNS text AS $$
  SELECT unaccent($1)
$$ LANGUAGE sql IMMUTABLE;

CREATE TABLE eventos (
    id                    UUID          PRIMARY KEY DEFAULT gen_random_uuid(),

    -- Campos principales
    tipo                  VARCHAR(30)   NOT NULL,
    titulo                VARCHAR(300)  NOT NULL,

    -- Fecha y hora
    fecha_inicio          DATE          NOT NULL,
    fecha_fin             DATE,
    hora_inicio           TIME,
    hora_fin              TIME,
    todo_el_dia           BOOLEAN       NOT NULL DEFAULT false,

    -- Localización y personas
    lugar                 VARCHAR(300),
    personas_relacionadas VARCHAR(500),

    -- Contenido
    descripcion           TEXT,
    recordatorio_minutos  INTEGER,

    -- Campos específicos por tipo (JSONB — flexible y rápido)
    datos_extra           JSONB         DEFAULT '{}',

    -- Auditoría de voz
    texto_original_voz    TEXT,

    -- Búsqueda full-text (columna generada automáticamente)
    texto_busqueda        TSVECTOR GENERATED ALWAYS AS (
        to_tsvector('spanish',
            unaccent_immutable(coalesce(titulo, ''))           || ' ' ||
            unaccent_immutable(coalesce(lugar, ''))            || ' ' ||
            unaccent_immutable(coalesce(descripcion, ''))      || ' ' ||
            unaccent_immutable(coalesce(personas_relacionadas, '')) || ' ' ||
            unaccent_immutable(coalesce(tipo, ''))
        )
    ) STORED,

    -- Timestamps
    created_at            TIMESTAMP     NOT NULL DEFAULT now(),
    updated_at            TIMESTAMP     NOT NULL DEFAULT now()
);

-- ── Índices ──────────────────────────────────────────────────
-- Full-text search (GIN — el más rápido para texto)
CREATE INDEX idx_eventos_busqueda   ON eventos USING GIN(texto_busqueda);

-- Consultas por fecha (el más usado en un calendario)
CREATE INDEX idx_eventos_fecha      ON eventos(fecha_inicio);
CREATE INDEX idx_eventos_fecha_rng  ON eventos(fecha_inicio, fecha_fin);

-- Filtros por tipo
CREATE INDEX idx_eventos_tipo       ON eventos(tipo);

-- Consultas JSONB (para campos específicos por tipo)
CREATE INDEX idx_eventos_jsonb      ON eventos USING GIN(datos_extra);

-- ── Comentarios de documentación ─────────────────────────────
COMMENT ON TABLE eventos IS 'Eventos del calendario inteligente FASI';
COMMENT ON COLUMN eventos.tipo IS 'CUMPLEANOS|CITA_MEDICA|TUTORIA|CLAUSTRO|CURSO|EXTRAESCOLAR|VIAJE|VACACIONES|COMIDA|COMPRAS|OTRO';
COMMENT ON COLUMN eventos.datos_extra IS 'Campos específicos por tipo en formato JSONB';
COMMENT ON COLUMN eventos.texto_original_voz IS 'Texto dictado por voz tal como lo reconoció el navegador';
COMMENT ON COLUMN eventos.texto_busqueda IS 'Vector full-text generado automáticamente para búsqueda rápida';
