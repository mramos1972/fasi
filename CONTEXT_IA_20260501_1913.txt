package com.miempresa.fasi.calendario;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.format.annotation.DateTimeFormat;
import org.springframework.http.ResponseEntity;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.List;
import java.util.UUID;

@Controller
@RequestMapping("/calendario")
@RequiredArgsConstructor
public class CalendarioController {

    private final EventoService    eventoService;
    private final EventoIaService  eventoIaService;

    // ── Vista principal (mes actual por defecto) ─────────────
    @GetMapping
    public String vistaCalendario(
            @RequestParam(required = false) Integer year,
            @RequestParam(required = false) Integer month,
            @RequestParam(required = false) String vista,
            Model model) {

        LocalDate hoy     = LocalDate.now();
        int       anio    = year  != null ? year  : hoy.getYear();
        int       mes     = month != null ? month : hoy.getMonthValue();
        String    vistaSel = vista != null ? vista : "mes";

        YearMonth ym      = YearMonth.of(anio, mes);

        model.addAttribute("eventos",     eventoService.findByMes(anio, mes));
        model.addAttribute("proximos",    eventoService.findProximos(5));
        model.addAttribute("tipos",       Evento.TipoEvento.values());
        model.addAttribute("yearMonth",   ym);
        model.addAttribute("hoy",         hoy);
        model.addAttribute("vista",       vistaSel);
        model.addAttribute("nuevoEvento", new Evento());

        return "calendario/index";
    }

    // ── Vista de un día concreto ─────────────────────────────
    @GetMapping("/dia/{fecha}")
    public String viaDia(@PathVariable @DateTimeFormat(iso = DateTimeFormat.ISO.DATE)
                         LocalDate fecha, Model model) {
        model.addAttribute("fecha",   fecha);
        model.addAttribute("eventos", eventoService.findByDia(fecha));
        model.addAttribute("tipos",   Evento.TipoEvento.values());
        return "calendario/dia";
    }

    // ── Formulario nuevo evento (manual) ─────────────────────
    @GetMapping("/nuevo")
    public String nuevoForm(@RequestParam(required = false)
                            @DateTimeFormat(iso = DateTimeFormat.ISO.DATE) LocalDate fecha,
                            Model model) {
        Evento e = new Evento();
        e.setFechaInicio(fecha != null ? fecha : LocalDate.now());
        model.addAttribute("evento", e);
        model.addAttribute("tipos",  Evento.TipoEvento.values());
        return "calendario/form";
    }

    // ── Formulario editar evento ─────────────────────────────
    @GetMapping("/{id}/editar")
    public String editarForm(@PathVariable UUID id, Model model) {
        model.addAttribute("evento", eventoService.findById(id));
        model.addAttribute("tipos",  Evento.TipoEvento.values());
        return "calendario/form";
    }

    // ── Guardar nuevo evento ─────────────────────────────────
    @PostMapping
    public String guardar(@Valid @ModelAttribute Evento evento) {
        eventoService.save(evento);
        return "redirect:/calendario?year=" + evento.getFechaInicio().getYear()
                + "&month=" + evento.getFechaInicio().getMonthValue();
    }

    // ── Actualizar evento ────────────────────────────────────
    @PostMapping("/{id}")
    public String actualizar(@PathVariable UUID id,
                             @Valid @ModelAttribute Evento evento) {
        eventoService.update(id, evento);
        return "redirect:/calendario?year=" + evento.getFechaInicio().getYear()
                + "&month=" + evento.getFechaInicio().getMonthValue();
    }

    // ── Eliminar evento ──────────────────────────────────────
    @PostMapping("/{id}/eliminar")
    public String eliminar(@PathVariable UUID id) {
        eventoService.delete(id);
        return "redirect:/calendario";
    }

    // ── API: Buscar eventos (buscador full-text) ─────────────
    @GetMapping("/api/buscar")
    @ResponseBody
    public ResponseEntity<List<Evento>> buscar(@RequestParam String q) {
        return ResponseEntity.ok(eventoService.buscar(q));
    }

    // ── API: Interpretar texto con IA → preview evento ───────
    @PostMapping("/api/interpretar")
    @ResponseBody
    public ResponseEntity<EventoIaResponse> interpretarVoz(
            @Valid @RequestBody EventoIaRequest request) {
        return ResponseEntity.ok(eventoIaService.interpretar(request));
    }

    // ── API: Confirmar y guardar evento interpretado por IA ──
    @PostMapping("/api/confirmar")
    @ResponseBody
    public ResponseEntity<Evento> confirmarEvento(@RequestBody EventoConfirmarRequest dto) {
        Evento evento = dto.toEvento();
        return ResponseEntity.ok(eventoService.save(evento));
    }

    // ── API: Consulta IA sobre el calendario ─────────────────
    @PostMapping("/api/consulta")
    @ResponseBody
    public ResponseEntity<String> consultaIa(@RequestBody String pregunta) {
        return ResponseEntity.ok(eventoIaService.consultar(pregunta));
    }

    // ── API: Eventos del mes (para renderizado dinámico) ─────
    @GetMapping("/api/mes")
    @ResponseBody
    public ResponseEntity<List<Evento>> eventosMes(
            @RequestParam int year, @RequestParam int month) {
        return ResponseEntity.ok(eventoService.findByMes(year, month));
    }
}

package com.miempresa.fasi.calendario;

import com.miempresa.fasi.common.BaseEntity;
import jakarta.persistence.*;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;
import lombok.*;
import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.Map;

@Entity
@Table(name = "eventos")
@Getter @Setter @NoArgsConstructor @AllArgsConstructor @Builder
public class Evento extends BaseEntity {

    // ── Tipo de evento ───────────────────────────────────────
    @NotNull
    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 30)
    private TipoEvento tipo;

    // ── Título generado por IA o escrito por el usuario ──────
    @NotBlank
    @Column(nullable = false, length = 300)
    private String titulo;

    // ── Fecha y hora ─────────────────────────────────────────
    @NotNull
    @Column(name = "fecha_inicio", nullable = false)
    private LocalDate fechaInicio;

    @Column(name = "fecha_fin")
    private LocalDate fechaFin;

    @Column(name = "hora_inicio")
    private LocalTime horaInicio;

    @Column(name = "hora_fin")
    private LocalTime horaFin;

    @Column(name = "todo_el_dia", nullable = false)
    private boolean todoDia = false;

    // ── Localización y personas ──────────────────────────────
    @Column(length = 300)
    private String lugar;

    @Column(name = "personas_relacionadas", length = 500)
    private String personasRelacionadas;

    // ── Contenido ────────────────────────────────────────────
    @Column(columnDefinition = "TEXT")
    private String descripcion;

    @Column(name = "recordatorio_minutos")
    private Integer recordatorioMinutos;

    // ── Datos específicos por tipo (JSONB) ───────────────────
    @JdbcTypeCode(SqlTypes.JSON)
    @Column(name = "datos_extra", columnDefinition = "jsonb")
    private Map<String, Object> datosExtra;

    // ── Auditoría de voz ─────────────────────────────────────
    @Column(name = "texto_original_voz", columnDefinition = "TEXT")
    private String textoOriginalVoz;

    // ── Enum de tipos ────────────────────────────────────────
    @com.fasterxml.jackson.annotation.JsonFormat(shape = com.fasterxml.jackson.annotation.JsonFormat.Shape.OBJECT)
    public enum TipoEvento {
        CUMPLEANOS,
        CITA_MEDICA,
        TUTORIA,
        CLAUSTRO,
        CURSO,
        EXTRAESCOLAR,
        VIAJE,
        VACACIONES,
        COMIDA,
        COMPRAS,
        OTRO;

        public String getEmoji() {
            return switch (this) {
                case CUMPLEANOS   -> "🎂";
                case CITA_MEDICA  -> "🏥";
                case TUTORIA      -> "📚";
                case CLAUSTRO     -> "🏫";
                case CURSO        -> "📖";
                case EXTRAESCOLAR -> "🎭";
                case VIAJE        -> "✈️";
                case VACACIONES   -> "🏖️";
                case COMIDA       -> "🍽️";
                case COMPRAS      -> "🛒";
                case OTRO         -> "⚙️";
            };
        }

        public String getColor() {
            return switch (this) {
                case CUMPLEANOS   -> "#f59e0b"; // amber
                case CITA_MEDICA  -> "#ef4444"; // red
                case TUTORIA      -> "#3b82f6"; // blue
                case CLAUSTRO     -> "#8b5cf6"; // violet
                case CURSO        -> "#06b6d4"; // cyan
                case EXTRAESCOLAR -> "#ec4899"; // pink
                case VIAJE        -> "#10b981"; // emerald
                case VACACIONES   -> "#f97316"; // orange
                case COMIDA       -> "#84cc16"; // lime
                case COMPRAS      -> "#6366f1"; // indigo
                case OTRO         -> "#6b7280"; // gray
            };
        }

        public String getLabel() {
            return switch (this) {
                case CUMPLEANOS   -> "Cumpleaños";
                case CITA_MEDICA  -> "Cita Médica";
                case TUTORIA      -> "Tutoría";
                case CLAUSTRO     -> "Claustro";
                case CURSO        -> "Curso";
                case EXTRAESCOLAR -> "Extraescolar";
                case VIAJE        -> "Viaje";
                case VACACIONES   -> "Vacaciones";
                case COMIDA       -> "Comida";
                case COMPRAS      -> "Compras";
                case OTRO         -> "Otro";
            };
        }

        /** Necesario para que el DTO pueda resolver el tipo por nombre */
        public String getName() { return this.name(); }

        @com.fasterxml.jackson.annotation.JsonCreator
        public static TipoEvento fromValue(Object value) {
            if (value instanceof String s) {
                try { return TipoEvento.valueOf(s.trim().toUpperCase()); }
                catch (IllegalArgumentException ignored) {}
                // Buscar por label
                for (TipoEvento t : values()) {
                    if (t.getLabel().equalsIgnoreCase(s.trim())) return t;
                }
            }
            return OTRO;
        }
    }
}

package com.miempresa.fasi.calendario;

import lombok.Data;
import java.time.LocalDate;
import java.time.LocalTime;
import java.util.Map;

/**
 * DTO para recibir el evento interpretado por la IA desde el frontend.
 * El campo 'tipo' puede llegar como String ("CUMPLEANOS") o como objeto
 * con subcampos {label, emoji, color} — lo normalizamos aquí.
 */
@Data
public class EventoConfirmarRequest {

    private Object tipo;   // puede ser String o Map<String,Object>

    private String    titulo;
    private LocalDate fechaInicio;
    private LocalDate fechaFin;
    private LocalTime horaInicio;
    private LocalTime horaFin;
    private boolean   todoDia;
    private String    lugar;
    private String    personasRelacionadas;
    private String    descripcion;
    private Integer   recordatorioMinutos;
    private Map<String, Object> datosExtra;
    private String    textoOriginalVoz;

    /**
     * Resuelve el tipo sea cual sea el formato que mande el frontend.
     */
    public Evento.TipoEvento resolverTipo() {
        if (tipo == null) return Evento.TipoEvento.OTRO;

        // Caso 1: llega como String puro → "CUMPLEANOS"
        if (tipo instanceof String s) {
            try { return Evento.TipoEvento.valueOf(s.trim().toUpperCase()); }
            catch (IllegalArgumentException e) { return Evento.TipoEvento.OTRO; }
        }

        // Caso 2: llega como objeto {label:"Cumpleaños", emoji:"🎂", color:"#f59e0b"}
        // Jackson lo deserializa como LinkedHashMap
        if (tipo instanceof Map<?,?> m) {
            // Intentar por nombre del enum si viene en el mapa
            Object nombre = m.get("name");
            if (nombre == null) nombre = m.get("nombre");
            if (nombre instanceof String s) {
                try { return Evento.TipoEvento.valueOf(s.trim().toUpperCase()); }
                catch (IllegalArgumentException ignored) {}
            }
            // Intentar por label
            Object label = m.get("label");
            if (label instanceof String lbl) {
                for (Evento.TipoEvento t : Evento.TipoEvento.values()) {
                    if (t.getLabel().equalsIgnoreCase(lbl.trim())) return t;
                }
            }
        }

        return Evento.TipoEvento.OTRO;
    }

    /** Convierte el DTO a entidad lista para persistir. */
    public Evento toEvento() {
        Evento e = new Evento();
        e.setTipo(resolverTipo());
        e.setTitulo(titulo != null ? titulo : "Sin título");
        e.setFechaInicio(fechaInicio != null ? fechaInicio : LocalDate.now());
        e.setFechaFin(fechaFin);
        e.setHoraInicio(horaInicio);
        e.setHoraFin(horaFin);
        e.setTodoDia(todoDia);
        e.setLugar(lugar);
        e.setPersonasRelacionadas(personasRelacionadas);
        e.setDescripcion(descripcion);
        e.setRecordatorioMinutos(recordatorioMinutos);
        e.setDatosExtra(datosExtra);
        e.setTextoOriginalVoz(textoOriginalVoz);
        return e;
    }
}

            String jsonLimpio = extraerJson(json);

            return new EventoIaResponse(true, null, evento,
                    System.currentTimeMillis() - inicio);

        } catch (Exception e) {

            return new EventoIaResponse(false,
                    "No pude interpretar el texto. Inténtalo de nuevo o crea el evento manualmente.",

                .findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(hoy, fin);

        if (proximos.isEmpty()) {
            ctx.append("(No hay eventos próximos registrados)\n");
        } else {
            ctx.append("Próximos 3 meses:\n");
            proximos.forEach(e -> ctx.append(String.format(

                        tipo.getEmoji(), tipo.getLabel(), count));
            }

        // Buscar primer { y último }
        int inicio = respuesta.indexOf('{');
        int fin    = respuesta.lastIndexOf('}');
        if (inicio >= 0 && fin > inicio) {
            return respuesta.substring(inicio, fin + 1);
        }
        return respuesta.trim();
    }

@Transactional(readOnly = true)

    // ── Lectura ──────────────────────────────────────────────

    public List<Evento> findByMes(int year, int month) {
        LocalDate inicio = LocalDate.of(year, month, 1);
        LocalDate fin    = inicio.withDayOfMonth(inicio.lengthOfMonth());
        return repo.findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(inicio, fin);
    }

    public List<Evento> findBySemana(LocalDate lunesSemana) {
        return repo.findByFechaInicioBetweenOrderByFechaInicioAscHoraInicioAsc(
                lunesSemana, lunesSemana.plusDays(6));
    }

    public List<Evento> findByDia(LocalDate fecha) {
        return repo.findByFechaInicioOrderByHoraInicioAsc(fecha);
    }

    public List<Evento> buscar(String query) {
        if (query == null || query.isBlank()) return List.of();
        return repo.buscarFullText(query.trim());
    }

        return repo.findProximos(LocalDate.now())
                   .stream().limit(limite).toList();

            .orElseThrow(() -> new EntityNotFoundException("Evento no encontrado: " + id));

import org.springframework.web.bind.annotation.*;

import java.util.stream.Collectors;

@Slf4j
@ControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(NoResourceFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    @ResponseBody
    public void noResource() { }

    @ExceptionHandler(EntityNotFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public String notFound(EntityNotFoundException ex, Model model) {
        model.addAttribute("error", ex.getMessage());
        return "error/404";
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    @ResponseBody
    public ApiResponse<Void> validationError(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + ": " + e.getDefaultMessage())
            .collect(Collectors.joining(", "));
        return ApiResponse.error(msg);
    }

    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public String genericError(Exception ex, Model model) {
        log.error("Error no controlado", ex);
        model.addAttribute("error", "Error interno del servidor");
        return "error/500";
    }
}

package com.miempresa.fasi.ia;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Slf4j
@Service
@RequiredArgsConstructor
public class IaService {

    private final OllamaClient       ollamaClient;
    private final IaContextBuilder   contextBuilder;
    private final IaMensajeRepository repo;

    @Transactional
    public IaChatResponse chat(IaChatRequest request) {
        long inicio = System.currentTimeMillis();

        String prompt    = contextBuilder.buildPrompt(request.getMensaje(), request.getModulo());
        String respuesta = ollamaClient.generate(prompt);
        long   tiempoMs  = System.currentTimeMillis() - inicio;

        // Persistir en historial
        repo.save(IaMensaje.builder()
                .pregunta(request.getMensaje())
                .respuesta(respuesta)
                .modulo(request.getModulo() != null ? request.getModulo() : "general")
                .tiempoMs(tiempoMs)
                .build());

        log.info("IA respondió en {}ms para módulo '{}'", tiempoMs, request.getModulo());
        return new IaChatResponse(respuesta, request.getModulo(), tiempoMs);
    }

    public List<IaMensaje> historial() {
        return repo.findTop20ByOrderByCreatedAtDesc();
    }
}

package com.miempresa.fasi.ia;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.Data;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

@Slf4j
@Component
@RequiredArgsConstructor
public class OllamaClient {

    @Value("${fasi.ollama.url:http://localhost:11434}")
    private String ollamaUrl;

    @Value("${fasi.ollama.model:gemma3:4b}")
    private String model;

    private final ObjectMapper objectMapper;
    private final HttpClient   httpClient = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(10))
            .build();

    public String generate(String prompt) {
        try {
            OllamaRequest req = new OllamaRequest();
            req.setModel(model);
            req.setPrompt(prompt);
            req.setStream(false);

            String body = objectMapper.writeValueAsString(req);

            HttpRequest httpReq = HttpRequest.newBuilder()
                    .uri(URI.create(ollamaUrl + "/api/generate"))
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(body))
                    .timeout(Duration.ofSeconds(120))
                    .build();

            HttpResponse<String> resp = httpClient.send(
                    httpReq, HttpResponse.BodyHandlers.ofString());

            OllamaResponse ollamaResp =
                    objectMapper.readValue(resp.body(), OllamaResponse.class);
            return ollamaResp.getResponse();

        } catch (Exception e) {
            log.error("Error llamando a Ollama", e);
            return "⚠️ Error al conectar con el asistente IA. " +
                   "Asegúrate de que Ollama está corriendo en WSL " +
                   "(ollama serve).";
        }
    }

    // ── DTOs internos ────────────────────────────────────────
    @Data
    static class OllamaRequest {
        private String  model;
        private String  prompt;
        private boolean stream;
    }

    @Data
    @JsonIgnoreProperties(ignoreUnknown = true)
    static class OllamaResponse {
        private String response;
        private boolean done;
    }
}

  <script th:inline="javascript">
    const EVENTOS_MES = /*[[${eventos}]]*/ [];
    const YEAR_MONTH  = /*[[${yearMonth.toString()}]]*/ '';
    const HOY         = /*[[${hoy.toString()}]]*/ '';

    // ── Renderizar grid ──────────────────────────────────────
    function renderCalendario() {
      const [year, month] = YEAR_MONTH.split('-').map(Number);
      const primerDia     = new Date(year, month - 1, 1);
      const ultimoDia     = new Date(year, month, 0).getDate();
      let diaSemana = primerDia.getDay() - 1;
      if (diaSemana < 0) diaSemana = 6;

      const grid = document.getElementById('grid-calendario');
      grid.innerHTML = '';

      // Celdas vacías
      for (let i = 0; i < diaSemana; i++) {
        grid.insertAdjacentHTML('beforeend',
          '<div class="min-h-28 border-b border-r border-gray-100 dark:border-gray-700 bg-gray-50 dark:bg-gray-900/20"></div>');
      }

      // Celdas de días
      for (let dia = 1; dia <= ultimoDia; dia++) {
        const fechaStr   = `${year}-${String(month).padStart(2,'0')}-${String(dia).padStart(2,'0')}`;
        const esHoy      = fechaStr === HOY;
        const eventosDia = EVENTOS_MES.filter(e => {
          // Evento puntual o inicio del rango
          if (e.fechaInicio === fechaStr) return true;
          // Evento con rango: pintar en todos los días intermedios y fin
          if (e.fechaFin && e.fechaFin !== e.fechaInicio) {
            const dInicio = new Date(e.fechaInicio + 'T00:00:00');
            const dFin    = new Date(e.fechaFin    + 'T00:00:00');
            const dCelda  = new Date(fechaStr       + 'T00:00:00');
            return dCelda > dInicio && dCelda <= dFin;
          }
          return false;
        });

        let html = `
          <div class="min-h-28 border-b border-r border-gray-100 dark:border-gray-700 p-1 cursor-pointer
                      hover:bg-indigo-50 dark:hover:bg-indigo-900/20 transition"
               onclick="window.location='/calendario/dia/${fechaStr}'">
            <div class="flex justify-between items-start mb-1">
              <span class="${esHoy
                ? 'bg-indigo-600 text-white w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold'
                : 'text-sm font-semibold text-gray-700 dark:text-gray-300'}">${dia}</span>
              <a href="/calendario/nuevo?fecha=${fechaStr}"
                 onclick="event.stopPropagation()"
                 class="text-gray-300 hover:text-indigo-500 text-lg leading-none">+</a>
            </div>`;

        eventosDia.slice(0, 3).forEach(e => {
          const label = (e.tipo && e.tipo.label) ? e.tipo.label : (e.tipo || '');
          const emoji = (e.tipo && e.tipo.emoji) ? e.tipo.emoji : '📅';
          const color = (e.tipo && e.tipo.color) ? e.tipo.color : '#6366f1';
          html += `
            <div class="text-xs px-1.5 py-0.5 rounded mb-0.5 truncate text-white font-semibold shadow-sm cursor-pointer"
                 style="background-color:${color}"
                 title="${e.titulo}"
                 onclick="event.stopPropagation(); window.location='/calendario/${e.id}/editar'">
              ${emoji} ${e.titulo}
            </div>`;
        });

        if (eventosDia.length > 3) {
          html += `<div class="text-xs text-gray-400 dark:text-gray-500">+${eventosDia.length - 3} más</div>`;
        }

        html += '</div>';
        grid.insertAdjacentHTML('beforeend', html);
      }
    }

    async function buscarEventos() {

    function toggleMicrofono() {
      grabando ? pararMicrofono() : iniciarMicrofono();
    }

    function iniciarMicrofono() {
      if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        alert('Tu navegador no soporta reconocimiento de voz. Usa Chrome.');
        return;
      }
      const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
      recognition = new SR();
      recognition.lang            = 'es-ES';
      recognition.continuous      = true;
      recognition.interimResults  = true;

      recognition.onresult = (event) => {
        const transcript = Array.from(event.results).map(r => r[0].transcript).join('');
        document.getElementById('voz-texto').value = transcript;
      };

      recognition.onend = () => {
        // Solo reiniciar si el usuario no paró manualmente Y hay silencio breve
        if (grabando) {
          setTimeout(() => {
            if (grabando) {
              try { recognition.start(); } catch(e) { /* ya iniciado */ }
            }
          }, 300);
        }
      };

      recognition.onerror = (e) => {
        console.warn('Speech error:', e.error);
        // 'no-speech' es normal (silencio), no parar
        if (e.error === 'no-speech') return;
        // Para cualquier otro error real, detener limpiamente
        grabando = false;
        actualizarBotonMicro(false);
      };

      recognition.start();
      grabando = true;
      actualizarBotonMicro(true);
    }

    function pararMicrofono() {
      grabando = false;
      if (recognition) recognition.stop();
      actualizarBotonMicro(false);
    }

    function actualizarBotonMicro(activo) {
      const btn   = document.getElementById('btn-micro');
      const label = document.getElementById('micro-label');
      if (activo) {
        label.textContent = '⏹ Parar grabación';
        btn.classList.add('bg-red-700');
        btn.classList.remove('bg-red-500');
      } else {
        label.textContent = '🎤 Iniciar grabación';
        btn.classList.remove('bg-red-700');
        btn.classList.add('bg-red-500');
      }
    }

    async function interpretarTexto() {
      const texto = document.getElementById('voz-texto').value.trim();
      if (!texto) { alert('Escribe o dicta algo primero.'); return; }

      const spinner = document.getElementById('ia-spinner');
      const preview = document.getElementById('preview-evento');
      spinner.style.display = 'flex';
      preview.style.display = 'none';

      try {
        const resp = await fetch('/calendario/api/interpretar', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify({ texto, fechaContexto: HOY })
        });
        const data = await resp.json();

        if (data.exito && data.eventoPreview) {
          eventoActual = data.eventoPreview;
          mostrarPreview(data.eventoPreview);
        } else {
          alert('❌ ' + (data.mensajeError || 'Error al interpretar'));
        }
      } catch (err) {
        alert('Error de conexión: ' + err.message);
      } finally {
        spinner.style.display = 'none';
      }
    }

    function mostrarPreview(evento) {
      const c = document.getElementById('preview-contenido');
      const tipoLabel = (evento.tipo && evento.tipo.label) ? evento.tipo.label : (evento.tipo || '');
      const tipoEmoji = (evento.tipo && evento.tipo.emoji) ? evento.tipo.emoji : '📅';
      c.innerHTML = `
        <p><strong>Tipo:</strong> ${tipoEmoji} ${tipoLabel}</p>
        <p><strong>Título:</strong> ${evento.titulo || '-'}</p>
        <p><strong>Fecha:</strong> ${evento.fechaInicio || '-'}${evento.horaInicio ? ' a las ' + evento.horaInicio : ''}</p>
        ${evento.lugar ? `<p><strong>Lugar:</strong> ${evento.lugar}</p>` : ''}
        ${evento.personasRelacionadas ? `<p><strong>Personas:</strong> ${evento.personasRelacionadas}</p>` : ''}
        ${evento.descripcion ? `<p><strong>Notas:</strong> ${evento.descripcion}</p>` : ''}
      `;
      document.getElementById('preview-evento').style.display = 'block';
    }

    async function confirmarEvento() {
      if (!eventoActual) return;
      try {
        const resp = await fetch('/calendario/api/confirmar', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify(eventoActual)
        });
        if (resp.ok) {
          cerrarModalVoz();
          window.location.reload();
        } else {
          alert('Error al guardar el evento.');
        }
      } catch (err) {
        alert('Error de conexión: ' + err.message);
      }
    }

    function editarManual() {
      if (!eventoActual) return;
      window.location = '/calendario/nuevo?fecha=' + (eventoActual.fechaInicio || HOY);
    }

-- ============================================================
--  FASI Calendario — V8
--  Tabla principal de eventos con búsqueda full-text
-- ============================================================

-- Extensión para búsqueda full-text en español
CREATE EXTENSION IF NOT EXISTS unaccent;

-- Wrapper inmutable de unaccent (necesario para columnas GENERATED)
CREATE OR REPLACE FUNCTION unaccent_immutable(text)
RETURNS text AS $$
  SELECT unaccent($1)
$$ LANGUAGE sql IMMUTABLE;

CREATE TABLE eventos (
    id                    UUID          PRIMARY KEY DEFAULT gen_random_uuid(),

    -- Campos principales
    tipo                  VARCHAR(30)   NOT NULL,
    titulo                VARCHAR(300)  NOT NULL,

    -- Fecha y hora
    fecha_inicio          DATE          NOT NULL,
    fecha_fin             DATE,
    hora_inicio           TIME,
    hora_fin              TIME,
    todo_el_dia           BOOLEAN       NOT NULL DEFAULT false,

    -- Localización y personas
    lugar                 VARCHAR(300),
    personas_relacionadas VARCHAR(500),

    -- Contenido
    descripcion           TEXT,
    recordatorio_minutos  INTEGER,

    -- Campos específicos por tipo (JSONB — flexible y rápido)
    datos_extra           JSONB         DEFAULT '{}',

    -- Auditoría de voz
    texto_original_voz    TEXT,

    -- Búsqueda full-text (columna generada automáticamente)
    texto_busqueda        TSVECTOR GENERATED ALWAYS AS (
        to_tsvector('spanish',
            unaccent_immutable(coalesce(titulo, ''))           || ' ' ||
            unaccent_immutable(coalesce(lugar, ''))            || ' ' ||
            unaccent_immutable(coalesce(descripcion, ''))      || ' ' ||
            unaccent_immutable(coalesce(personas_relacionadas, '')) || ' ' ||
            unaccent_immutable(coalesce(tipo, ''))
        )
    ) STORED,

    -- Timestamps
    created_at            TIMESTAMP     NOT NULL DEFAULT now(),
    updated_at            TIMESTAMP     NOT NULL DEFAULT now()
);

-- ── Índices ──────────────────────────────────────────────────
-- Full-text search (GIN — el más rápido para texto)
CREATE INDEX idx_eventos_busqueda   ON eventos USING GIN(texto_busqueda);

-- Consultas por fecha (el más usado en un calendario)
CREATE INDEX idx_eventos_fecha      ON eventos(fecha_inicio);
CREATE INDEX idx_eventos_fecha_rng  ON eventos(fecha_inicio, fecha_fin);

-- Filtros por tipo
CREATE INDEX idx_eventos_tipo       ON eventos(tipo);

-- Consultas JSONB (para campos específicos por tipo)
CREATE INDEX idx_eventos_jsonb      ON eventos USING GIN(datos_extra);

-- ── Comentarios de documentación ─────────────────────────────
COMMENT ON TABLE eventos IS 'Eventos del calendario inteligente FASI';
COMMENT ON COLUMN eventos.tipo IS 'CUMPLEANOS|CITA_MEDICA|TUTORIA|CLAUSTRO|CURSO|EXTRAESCOLAR|VIAJE|VACACIONES|COMIDA|COMPRAS|OTRO';
COMMENT ON COLUMN eventos.datos_extra IS 'Campos específicos por tipo en formato JSONB';
COMMENT ON COLUMN eventos.texto_original_voz IS 'Texto dictado por voz tal como lo reconoció el navegador';
COMMENT ON COLUMN eventos.texto_busqueda IS 'Vector full-text generado automáticamente para búsqueda rápida';
